// Valida y limpia los datos de una llamada (crear o editar).
export function parseCall(body: any, partial = false) {
  const out: any = {}
  const err = (m: string) => ({ error: m })

  if (!partial || body.title !== undefined) {
    const title = typeof body.title === 'string' ? body.title.trim() : ''
    if (!title) return err('Escribe el título de la llamada.')
    out.title = title.slice(0, 140)
  }
  if (!partial || body.description !== undefined) {
    out.description = typeof body.description === 'string' && body.description.trim() ? body.description.trim().slice(0, 1500) : null
  }
  if (!partial || body.startsAt !== undefined) {
    const d = new Date(body.startsAt)
    if (!body.startsAt || isNaN(d.getTime())) return err('Elige la fecha y hora de la llamada.')
    out.startsAt = d
  }
  if (!partial || body.durationMin !== undefined) {
    const n = parseInt(body.durationMin, 10)
    out.durationMin = Math.min(Math.max(isNaN(n) ? 60 : n, 5), 600)
  }
  if (!partial || body.joinUrl !== undefined) {
    const url = typeof body.joinUrl === 'string' ? body.joinUrl.trim() : ''
    try {
      const u = new URL(url)
      if (u.protocol !== 'https:') throw new Error()
    } catch {
      return err('El enlace debe ser una dirección válida que empiece con https:// (por ejemplo, el enlace de Zoom).')
    }
    out.joinUrl = url
  }
  if (!partial || body.meetingId !== undefined) {
    out.meetingId = typeof body.meetingId === 'string' && body.meetingId.trim() ? body.meetingId.trim().slice(0, 40) : null
  }
  if (!partial || body.passcode !== undefined) {
    out.passcode = typeof body.passcode === 'string' && body.passcode.trim() ? body.passcode.trim().slice(0, 40) : null
  }
  if (!partial || body.allowedRole !== undefined) {
    out.allowedRole = ACCESS_LEVELS.includes(body.allowedRole) ? body.allowedRole : 'ALL'
  }
  if (!partial || body.recordingUrl !== undefined) {
    const r = typeof body.recordingUrl === 'string' ? body.recordingUrl.trim() : ''
    if (r) {
      try {
        if (new URL(r).protocol !== 'https:') throw new Error()
      } catch {
        return err('El enlace de la grabación debe empezar con https://')
      }
      out.recordingUrl = r
    } else out.recordingUrl = null
  }
  if (!partial || body.isPublished !== undefined) {
    out.isPublished = body.isPublished !== false
  }
  return { data: out }
}

import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prismaDash } from '@/lib/prisma-dashboard'

export interface Actor {
  id: string
  name: string
  email: string
  role: string
}

export const MAX_LEN = 500

// Usuario autenticado y habilitado (aprobado y sin bloqueo general).
export async function getActor(): Promise<Actor | null> {
  const session: any = await getServerSession(authOptions)
  const id = session?.user?.id
  if (!id) return null
  const u = await prismaDash.user.findUnique({
    where: { id },
    select: { id: true, name: true, email: true, role: true, status: true, isBanned: true },
  })
  if (!u || u.isBanned || u.status !== 'APPROVED') return null
  return { id: u.id, name: u.name || u.email.split('@')[0], email: u.email, role: u.role }
}

export async function getAdmin(): Promise<Actor | null> {
  const a = await getActor()
  return a && a.role === 'ADMIN' ? a : null
}

// Sanción vigente del chat (silencio temporal o bloqueo). Limpia las vencidas.
export async function activeSanction(userId: string) {
  const s = await prismaDash.chatSanction.findUnique({ where: { userId } })
  if (!s) return null
  if (s.until && s.until.getTime() <= Date.now()) {
    await prismaDash.chatSanction.delete({ where: { userId } }).catch(() => {})
    return null
  }
  return s
}

export function cleanText(input: unknown): string {
  if (typeof input !== 'string') return ''
  return input
    .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F]/g, '')
    .replace(/\r\n/g, '\n')
    .replace(/\n{3,}/g, '\n\n')
    .trim()
}

// Huella anónima del visitante (no se guarda la IP real) para limitar abusos.
export function clientIpHash(req: Request): string {
  const h = req.headers
  const ip =
    h.get('x-nf-client-connection-ip') ||
    h.get('x-forwarded-for')?.split(',')[0]?.trim() ||
    h.get('x-real-ip') ||
    'unknown'
  return crypto
    .createHash('sha256')
    .update(ip + (process.env.NEXTAUTH_SECRET || 'invictus'))
    .digest('hex')
    .slice(0, 32)
}

export function short(input: unknown, max: number): string {
  if (typeof input !== 'string') return ''
  return input.replace(/[\u0000-\u001F\u007F]/g, ' ').replace(/\s+/g, ' ').trim().slice(0, max)
}

import { getAdmin, cleanText } from '@/lib/chat-utils'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

export async function GET() {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    const notices = await prismaDash.botNotice.findMany({ orderBy: { createdAt: 'desc' }, take: 50 })
    return NextResponse.json({ notices }, { headers: { 'Cache-Control': 'no-store' } })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

export async function POST(req: Request) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    const body = await req.json().catch(() => ({}))
    const content = cleanText(body?.content)
    if (!content) return NextResponse.json({ error: 'Escribe el aviso.' }, { status: 400 })
    if (content.length > 300) return NextResponse.json({ error: 'El aviso no puede superar los 300 caracteres.' }, { status: 400 })
    const active = await prismaDash.botNotice.count({ where: { isActive: true } })
    if (active >= 5) return NextResponse.json({ error: 'Máximo 5 avisos activos. Desactiva o elimina uno.' }, { status: 400 })
    const notice = await prismaDash.botNotice.create({ data: { content } })
    return NextResponse.json({ notice })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

export async function PATCH(req: Request) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    const { id, isActive } = await req.json().catch(() => ({}))
    if (typeof id !== 'string' || typeof isActive !== 'boolean') return NextResponse.json({ error: 'Datos inválidos.' }, { status: 400 })
    if (isActive) {
      const active = await prismaDash.botNotice.count({ where: { isActive: true, NOT: { id } } })
      if (active >= 5) return NextResponse.json({ error: 'Máximo 5 avisos activos.' }, { status: 400 })
    }
    await prismaDash.botNotice.update({ where: { id }, data: { isActive } })
    return NextResponse.json({ success: true })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

export async function DELETE(req: Request) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    const id = new URL(req.url).searchParams.get('id')
    if (!id) return NextResponse.json({ error: 'Falta el aviso.' }, { status: 400 })
    await prismaDash.botNotice.deleteMany({ where: { id } })
    return NextResponse.json({ success: true })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import { prismaDash } from '@/lib/prisma-dashboard'
import { getAdmin } from '@/lib/chat-utils'
import { avatarIds } from '@/lib/chat-avatars'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

// Buscar alumnos por nombre o correo para silenciarlos o bloquearlos del chat.
export async function GET(req: Request) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const q = (new URL(req.url).searchParams.get('q') || '').trim().slice(0, 60)
    if (q.length < 2) return NextResponse.json({ members: [], avatars: [] })

    const users = await prismaDash.user.findMany({
      where: {
        status: 'APPROVED',
        role: { not: 'ADMIN' },
        OR: [{ name: { contains: q, mode: 'insensitive' } }, { email: { contains: q, mode: 'insensitive' } }],
      },
      select: { id: true, name: true, email: true, role: true },
      orderBy: { name: 'asc' },
      take: 15,
    })
    const ids = users.map((u: any) => u.id)
    const sanctions = ids.length
      ? await prismaDash.chatSanction.findMany({
          where: { userId: { in: ids }, OR: [{ until: null }, { until: { gt: new Date() } }] },
        })
      : []
    const kindBy = new Map<string, string>(sanctions.map((s: any) => [s.userId, s.kind]))
    const avatars = await avatarIds(ids)

    return NextResponse.json(
      { members: users.map((u: any) => ({ ...u, sanction: kindBy.get(u.id) || null })), avatars },
      { headers: { 'Cache-Control': 'no-store' } }
    )
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

// Silenciar (MUTE, con duración en minutos) o bloquear (BAN) a un alumno del chat.
export async function POST(req: Request) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const body = await req.json().catch(() => ({}))
    const userId = typeof body.userId === 'string' ? body.userId : ''
    const kind = body.kind === 'BAN' ? 'BAN' : body.kind === 'MUTE' ? 'MUTE' : ''
    const reason = typeof body.reason === 'string' ? body.reason.trim().slice(0, 200) || null : null
    const purge = body.purge === true
    if (!userId || !kind) return NextResponse.json({ error: 'Faltan datos.' }, { status: 400 })

    if (userId === admin.id) return NextResponse.json({ error: 'No puedes sancionarte a ti mismo.' }, { status: 400 })

    const target = await prismaDash.user.findUnique({ where: { id: userId }, select: { id: true, role: true } })
    if (!target) return NextResponse.json({ error: 'El usuario no existe.' }, { status: 404 })
    if (target.role === 'ADMIN') return NextResponse.json({ error: 'No puedes sancionar a otro administrador.' }, { status: 400 })

    let until: Date | null = null
    if (kind === 'MUTE') {
      const minutes = Math.min(Math.max(parseInt(body.minutes, 10) || 60, 1), 60 * 24 * 30)
      until = new Date(Date.now() + minutes * 60000)
    }

    await prismaDash.chatSanction.upsert({
      where: { userId },
      update: { kind, reason, until, createdBy: admin.id, createdAt: new Date() },
      create: { userId, kind, reason, until, createdBy: admin.id },
    })

    if (purge) {
      await prismaDash.chatMessage.updateMany({
        where: { userId, deletedAt: null },
        data: { deletedAt: new Date(), deletedBy: admin.id, pinned: false },
      })
    }
    return NextResponse.json({ success: true })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

// Feed completo para moderar (incluye mensajes eliminados) + sanciones vigentes.
export async function GET() {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401, headers: NO_STORE })

    const messages = await prismaDash.chatMessage.findMany({ orderBy: { createdAt: 'desc' }, take: 150 })
    const now = new Date()
    const sanctions = await prismaDash.chatSanction.findMany({
      where: { OR: [{ until: null }, { until: { gt: now } }] },
      orderBy: { createdAt: 'desc' },
    })

    const ids = Array.from(new Set([...messages.map((m: any) => m.userId), ...sanctions.map((s: any) => s.userId)]))
    const users = ids.length
      ? await prismaDash.user.findMany({ where: { id: { in: ids } }, select: { id: true, email: true, name: true, role: true } })
      : []
    const byId = new Map<string, any>(users.map((u: any) => [u.id, u]))
    const sanctionByUser = new Map<string, any>(sanctions.map((s: any) => [s.userId, s]))

    const avatars = await avatarIds(ids)

    return NextResponse.json(
      {
        avatars,
        messages: messages.reverse().map((m: any) => ({
          ...m,
          userEmail: byId.get(m.userId)?.email || null,
          sanction: sanctionByUser.get(m.userId)?.kind || null,
        })),
        sanctions: sanctions.map((s: any) => ({
          id: s.id,
          userId: s.userId,
          kind: s.kind,
          reason: s.reason,
          until: s.until,
          createdAt: s.createdAt,
          userName: byId.get(s.userId)?.name || null,
          userEmail: byId.get(s.userId)?.email || null,
        })),
      },
      { headers: NO_STORE }
    )
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500, headers: NO_STORE })
  }
}

// Publicar un mensaje del equipo (opcionalmente fijado).
export async function POST(req: Request) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const body = await req.json().catch(() => ({}))
    const content = cleanText(body?.content)
    if (!content) return NextResponse.json({ error: 'Escribe un mensaje.' }, { status: 400 })
    if (content.length > MAX_LEN * 2) return NextResponse.json({ error: 'El mensaje es demasiado largo.' }, { status: 400 })

    const pinned = body?.pinned === true
    if (pinned && (await prismaDash.chatMessage.count({ where: { pinned: true, deletedAt: null } })) >= MAX_PINNED) {
      return NextResponse.json({ error: `Ya hay ${MAX_PINNED} mensajes fijos. Quita uno para fijar otro.` }, { status: 400 })
    }

    const message = await prismaDash.chatMessage.create({
      data: { userId: admin.id, userName: admin.name, userRole: 'ADMIN', content, pinned },
    })
    return NextResponse.json({ message })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

// Acciones sobre un mensaje: delete | restore | pin | unpin
export async function PATCH(req: Request) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const { id, action } = await req.json().catch(() => ({}))
    if (!id || typeof id !== 'string') return NextResponse.json({ error: 'Falta el mensaje.' }, { status: 400 })

    const msg = await prismaDash.chatMessage.findUnique({ where: { id }, select: { id: true } })
    if (!msg) return NextResponse.json({ error: 'El mensaje no existe.' }, { status: 404 })

    if (action === 'delete') {
      await prismaDash.chatMessage.update({ where: { id }, data: { deletedAt: new Date(), deletedBy: admin.id, pinned: false } })
    } else if (action === 'restore') {
      await prismaDash.chatMessage.update({ where: { id }, data: { deletedAt: null, deletedBy: null } })
    } else if (action === 'pin') {
      const already = await prismaDash.chatMessage.count({ where: { pinned: true, deletedAt: null, NOT: { id } } })
      if (already >= MAX_PINNED) {
        return NextResponse.json({ error: `Ya hay ${MAX_PINNED} mensajes fijos. Quita uno para fijar otro.` }, { status: 400 })
      }
      await prismaDash.chatMessage.update({ where: { id }, data: { pinned: true, deletedAt: null, deletedBy: null } })
    } else if (action === 'unpin') {
      await prismaDash.chatMessage.update({ where: { id }, data: { pinned: false } })
    } else {
      return NextResponse.json({ error: 'Acción no válida.' }, { status: 400 })
    }
    return NextResponse.json({ success: true })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

import { getServerSession } from 'next-auth'

import { parseUA } from '@/lib/useragent'
import { lookupGeo } from '@/lib/geo'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

export async function GET() {
  try {
    const s: any = await getServerSession(authOptions)
    if (!s?.user || s.user.role !== 'ADMIN') return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const now = Date.now()
    const onlineSince = new Date(now - 3 * 60 * 1000)
    const daySince = new Date(now - 24 * 3600 * 1000)

    const sessions = await prismaDash.session.findMany({
      where: { lastSeen: { gte: daySince }, token: { startsWith: 'web-' } },
      orderBy: { lastSeen: 'desc' },
      take: 300,
      include: { user: { select: { id: true, name: true, email: true, role: true, avatarUrl: true } } },
    })

    const ips = Array.from(new Set(sessions.map((x) => x.ip).filter(Boolean) as string[])).slice(0, 60)
    const geoMap = new Map<string, Awaited<ReturnType<typeof lookupGeo>>>()
    await Promise.all(ips.map(async (ip) => geoMap.set(ip, await lookupGeo(ip))))

    const lessonIds = Array.from(new Set(sessions.map((x) => x.currentLesson).filter(Boolean) as string[]))
    const events = await prismaDash.activityLog.findMany({
      orderBy: { createdAt: 'desc' },
      take: 50,
      include: { user: { select: { name: true, email: true, role: true } }, lesson: { select: { title: true } } },
    })
    const lessons = lessonIds.length
      ? await prismaDash.lesson.findMany({ where: { id: { in: lessonIds } }, select: { id: true, title: true } })
      : []
    const lessonTitle = new Map(lessons.map((l) => [l.id, l.title]))

    const rows = sessions.map((x) => {
      const ua = parseUA(x.userAgent)
      const g = x.ip ? geoMap.get(x.ip) ?? null : null
      return {
        id: x.id,
        online: x.lastSeen >= onlineSince,
        lastSeen: x.lastSeen,
        page: x.currentPage,
        lesson: x.currentLesson ? lessonTitle.get(x.currentLesson) ?? null : null,
        ip: x.ip,
        ...ua,
        country: g?.country ?? null,
        countryCode: g?.countryCode ?? null,
        city: g?.city ?? null,
        lat: g?.lat ?? null,
        lon: g?.lon ?? null,
        user: x.user,
      }
    })

    const count = (arr: (string | null)[]) => {
      const m = new Map<string, number>()
      arr.forEach((k) => { const key = k || 'Desconocido'; m.set(key, (m.get(key) || 0) + 1) })
      return Array.from(m, ([name, value]) => ({ name, value })).sort((a, b) => b.value - a.value)
    }

    const security = await prismaDash.securityLog.findMany({ orderBy: { createdAt: 'desc' }, take: 40 })
    const uniqueUsers24 = new Set(sessions.map((x) => x.userId)).size
    const onlineUsers = new Set(rows.filter((r) => r.online).map((r) => r.user.id)).size

    return NextResponse.json({
      stats: { onlineUsers, uniqueUsers24, events: events.length, alerts: security.length },
      sessions: rows,
      browsers: count(rows.map((r) => r.browser)),
      systems: count(rows.map((r) => r.os)),
      devices: count(rows.map((r) => r.device)),
      countries: count(rows.map((r) => r.country)),
      events: events.map((e) => ({
        id: e.id, action: e.action, page: e.page, createdAt: e.createdAt, ip: e.ip,
        user: e.user, lesson: e.lesson?.title ?? null,
      })),
      security: security.map((x) => ({ id: x.id, action: x.action, email: x.email, details: x.details, ip: x.ipAddress, createdAt: x.createdAt })),
    })

export async function PATCH(req: Request) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const body = await req.json()
    const id = body.id
    if (!id || typeof id !== 'string') {
      return NextResponse.json({ error: 'Falta el id del usuario' }, { status: 400 })
    }

    const target = await prismaDash.user.findUnique({ where: { id } })
    if (!target) return NextResponse.json({ error: 'Usuario no encontrado' }, { status: 404 })

    const isSelf = (!!admin.id && admin.id === target.id) || (!!admin.email && admin.email === target.email)

    const data: Record<string, any> = {}

    if (typeof body.role === 'string') {
      if (!ROLES.includes(body.role)) return NextResponse.json({ error: 'Rol no válido' }, { status: 400 })
      if (isSelf && body.role !== 'ADMIN') {
        return NextResponse.json({ error: 'No puedes quitarte tu propio rol de administrador' }, { status: 400 })
      }
      data.role = body.role
    }

    if (typeof body.status === 'string') {
      if (!STATUSES.includes(body.status)) return NextResponse.json({ error: 'Estado no válido' }, { status: 400 })
      if (isSelf && body.status !== 'APPROVED') {
        return NextResponse.json({ error: 'No puedes cambiar tu propio estado de aprobación' }, { status: 400 })
      }
      data.status = body.status
    }

    if (typeof body.isBanned === 'boolean') {
      if (isSelf && body.isBanned) {
        return NextResponse.json({ error: 'No puedes banearte a ti mismo' }, { status: 400 })
      }
      data.isBanned = body.isBanned
    }

    if (typeof body.name === 'string') data.name = body.name.trim() || null
    if (typeof body.phone === 'string') data.phone = body.phone.trim() || null

    if (Object.keys(data).length === 0) {
      return NextResponse.json({ error: 'No hay cambios para guardar' }, { status: 400 })
    }

    const user = await prismaDash.user.update({
      where: { id },
      data,
      select: {
        id: true, name: true, email: true, role: true, status: true,
        isBanned: true, phone: true, registeredIp: true, createdAt: true,
      },
    })

    // Si se banea o rechaza, cerrar sus sesiones activas
    if (data.isBanned === true || data.status === 'REJECTED') {
      await prismaDash.session.updateMany({
        where: { userId: id, isActive: true },
        data: { isActive: false },
      })
    }

    return NextResponse.json({ success: true, user })
  } catch (err: any) {
    console.error('PATCH students:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

const BUCKET = 'videos'
const MAX_MB = 50
const ALLOWED_EXT = ['mp4', 'webm', 'mkv', 'mov']

export async function POST(req: Request) {
  try {
    const session: any = await getServerSession(authOptions)
    if (!session?.user || session.user.role !== 'ADMIN') {
      return NextResponse.json({ error: 'No autorizado: tu sesión no es de administrador. Cierra sesión y vuelve a entrar.' }, { status: 401 })
    }

    const { fileName, fileSize } = await req.json()
    if (!fileName || typeof fileName !== 'string') {
      return NextResponse.json({ error: 'Falta el nombre del archivo.' }, { status: 400 })
    }

    const ext = fileName.split('.').pop()?.toLowerCase() || ''
    if (!ALLOWED_EXT.includes(ext)) {
      return NextResponse.json({ error: `Formato .${ext} no admitido. Usa MP4, WebM, MKV o MOV.` }, { status: 400 })
    }
    if (typeof fileSize === 'number' && fileSize > MAX_MB * 1024 * 1024) {
      return NextResponse.json(
        { error: `El video pesa ${(fileSize / 1024 / 1024).toFixed(0)} MB y el máximo es ${MAX_MB} MB.` },
        { status: 413 }
      )
    }

    const supabaseUrl = (process.env.SUPABASE_URL || process.env.NEXT_PUBLIC_SUPABASE_URL || '').trim()
    const serviceKey = (process.env.SUPABASE_SERVICE_ROLE_KEY || '').trim()
    const missing: string[] = []
    if (!supabaseUrl) missing.push('SUPABASE_URL')
    if (!serviceKey) missing.push('SUPABASE_SERVICE_ROLE_KEY')
    if (missing.length > 0) {
      return NextResponse.json(
        { error: `Falta configurar en Netlify: ${missing.join(', ')}. Agrégala en Environment variables y haz un redeploy.` },
        { status: 500 }
      )
    }

    const supabase = createClient(supabaseUrl, serviceKey, { auth: { persistSession: false } })

    const safeName = fileName.replace(/[^a-zA-Z0-9._-]/g, '_')
    const path = `${Date.now()}-${safeName}`

    const { data, error } = await supabase.storage.from(BUCKET).createSignedUploadUrl(path)
    if (error || !data) {
      const m = error?.message || ''
      const friendly = /bucket/i.test(m) && /not found/i.test(m)
        ? 'No existe el bucket "videos" en Supabase Storage. Créalo (público) con ese nombre exacto.'
        : `Supabase no pudo preparar la subida: ${m || 'error desconocido'}`
      return NextResponse.json({ error: friendly }, { status: 500 })
    }

    const { data: pub } = supabase.storage.from(BUCKET).getPublicUrl(path)

    return NextResponse.json({ success: true, path, token: data.token, url: pub.publicUrl })
  } catch (err: any) {
    console.error('upload:', err)
    return NextResponse.json({ error: err?.message || 'Error interno del servidor' }, { status: 500 })
  }

const GRACE_MIN = 60 // y sigue visible 60 min después de terminar

// Llamadas visibles según el nivel del alumno. El enlace de Zoom solo se entrega cuando corresponde.
export async function GET() {
  try {
    const me = await getActor()
    if (!me) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const levels = visibleLevels(me.role)
    const isAdmin = me.role === 'ADMIN'
    const where: any = { isPublished: true }
    if (levels) where.allowedRole = { in: levels }

    const now = Date.now()
    const since = new Date(now - 24 * 3600 * 1000)

    const [upcomingRaw, pastRaw] = await Promise.all([
      prismaDash.zoomCall.findMany({ where: { ...where, startsAt: { gte: since } }, orderBy: { startsAt: 'asc' }, take: 50 }),
      prismaDash.zoomCall.findMany({ where: { ...where, startsAt: { lt: new Date(now) } }, orderBy: { startsAt: 'desc' }, take: 40 }),
    ])

      const start = c.startsAt.getTime()
      const end = start + c.durationMin * 60000
      const unlockAt = start - UNLOCK_MIN * 60000
      const finished = now > end
      const status = finished ? 'PAST' : now >= start ? 'LIVE' : now >= unlockAt ? 'SOON' : 'UPCOMING'
      const linkOpen = isAdmin || (now >= unlockAt && now <= end + GRACE_MIN * 60000)

        id: c.id,
        title: c.title,
        description: c.description,
        startsAt: c.startsAt,
        durationMin: c.durationMin,
        allowedRole: c.allowedRole,
        status,
        unlockAt: new Date(unlockAt),
        joinUrl: linkOpen ? c.joinUrl : null,
        meetingId: linkOpen ? c.meetingId : null,
        passcode: linkOpen ? c.passcode : null,
        recordingUrl: finished ? c.recordingUrl : null,
      }
    }

    const upcoming = upcomingRaw.map(shape).filter((c: any) => c.status !== 'PAST')
    const past = pastRaw.map(shape).filter((c: any) => c.status === 'PAST')

    return NextResponse.json({ upcoming, past, now: new Date(now) }, { headers: { 'Cache-Control': 'no-store' } })

    }

    const body = await req.json().catch(() => ({}))

        where: { userId: me.id },
        orderBy: { createdAt: 'desc' },
        select: { content: true, createdAt: true },

        }
      }

      data: { userId: me.id, userName: me.name, userRole: me.role, content },

export async function PATCH(req: Request) {
  try {
    const session: any = await getServerSession(authOptions)
    const id = session?.user?.id
    if (!id) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const body = await req.json()
    const data: Record<string, any> = {}

    if (typeof body.name === 'string') {
      const n = body.name.trim()
      if (!n) return NextResponse.json({ error: 'El nombre no puede estar vacío' }, { status: 400 })
      data.name = n.slice(0, 80)
    }
    if (typeof body.phone === 'string') data.phone = body.phone.trim().slice(0, 30) || null

    if (body.avatarUrl === null) {
      data.avatarUrl = null
    } else if (typeof body.avatarUrl === 'string') {
      if (!AVATAR_RE.test(body.avatarUrl) || body.avatarUrl.length > MAX_AVATAR_CHARS) {
        return NextResponse.json({ error: 'La imagen no es válida o es demasiado pesada' }, { status: 400 })
      }
      data.avatarUrl = body.avatarUrl
    }

    if (typeof body.newPassword === 'string' && body.newPassword) {
      if (body.newPassword.length < 8) {
        return NextResponse.json({ error: 'La nueva contraseña debe tener al menos 8 caracteres' }, { status: 400 })
      }
      const current = await prismaDash.user.findUnique({ where: { id }, select: { password: true } })
      const ok = current && typeof body.currentPassword === 'string' && (await bcrypt.compare(body.currentPassword, current.password))
      if (!ok) return NextResponse.json({ error: 'La contraseña actual no es correcta' }, { status: 400 })
      data.password = await bcrypt.hash(body.newPassword, 10)
    }

    if (Object.keys(data).length === 0) {
      return NextResponse.json({ error: 'No hay cambios para guardar' }, { status: 400 })
    }

    const user = await prismaDash.user.update({
      where: { id },
      data,
      select: { name: true, email: true, phone: true, role: true, avatarUrl: true },
    })
    return NextResponse.json({ success: true, user })
  } catch (err: any) {
    console.error('PATCH student/profile:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

// El reproductor informa una señal sospechosa. El servidor decide si corresponde bloquear.
export async function POST(req: Request) {
  try {
    const s: any = await getServerSession(authOptions)
    if (!s?.user?.id) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    if (s.user.role === 'ADMIN') return NextResponse.json({ locked: false })

    const body = await req.json().catch(() => ({}))
    const signal = typeof body.signal === 'string' ? body.signal : ''
    const def = SIGNALS[signal]
    if (!def) return NextResponse.json({ error: 'Señal no válida' }, { status: 400 })

    const userId: string = s.user.id
    const lessonId = typeof body.lessonId === 'string' ? body.lessonId.slice(0, 60) : null

    const existing = await prismaDash.videoLock.findUnique({ where: { userId } })
    if (existing?.locked) return NextResponse.json({ locked: true, reason: existing.reason })

    // Evita registrar la misma señal repetida en pocos segundos
    const recentSame = await prismaDash.videoIncident.findFirst({
      where: { userId, signal, createdAt: { gt: new Date(Date.now() - 3000) } },
      select: { id: true },
    })
    if (recentSame) return NextResponse.json({ locked: false })

    let lessonTitle: string | null = null
    if (lessonId) {
      const l = await prismaDash.lesson.findUnique({ where: { id: lessonId }, select: { title: true } }).catch(() => null)
      lessonTitle = l?.title || null
    }

    const fwd = req.headers.get('x-forwarded-for')
    const ip = fwd ? fwd.split(',')[0].trim() : req.headers.get('x-real-ip') || null

    await prismaDash.videoIncident.create({
      data: {
        userId,
        userName: s.user.name || null,
        userEmail: s.user.email || null,
        lessonId,
        lessonTitle,
        signal,
        score: def.score,
        ip,
        userAgent: (req.headers.get('user-agent') || '').slice(0, 300) || null,
      },
    })

    // Suma de puntos desde el último desbloqueo (o los últimos 10 minutos)
    const since = new Date(Math.max(Date.now() - WINDOW_MS, existing?.unlockedAt ? existing.unlockedAt.getTime() : 0))
    const agg = await prismaDash.videoIncident.aggregate({
      where: { userId, createdAt: { gt: since } },
      _sum: { score: true },
    })
    const total = agg._sum.score || 0

    if (total >= LOCK_THRESHOLD) {
      const reason = signalLabel(signal)
      await prismaDash.videoLock.upsert({
        where: { userId },
        create: { userId, userName: s.user.name || null, userEmail: s.user.email || null, locked: true, reason, lessonId, lessonTitle },
        update: { userName: s.user.name || null, userEmail: s.user.email || null, locked: true, reason, lessonId, lessonTitle, lockedAt: new Date() },
      })
      return NextResponse.json({ locked: true, reason })
    }
    return NextResponse.json({ locked: false })
  } catch (err: any) {
    console.error('POST student/security:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

    setSel(null)
    loadThreads()
  }

  async function addNotice() {
    if (!newNotice.trim()) return
    setErr('')
    const r = await fetch('/api/admin/bot-notices', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ content: newNotice.trim() }) })
    const d = await r.json().catch(() => ({}))
    if (!r.ok) return setErr(d.error || 'No se pudo publicar')
    setNewNotice('')
    loadNotices()
  }
  async function toggleNotice(n: Notice) {
    setErr('')
    const r = await fetch('/api/admin/bot-notices', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: n.id, isActive: !n.isActive }) })
    if (!r.ok) setErr((await r.json().catch(() => ({}))).error || 'Error')
    loadNotices()
  }
  async function delNotice(n: Notice) {
    if (!confirm('¿Eliminar este aviso?')) return
    await fetch('/api/admin/bot-notices?id=' + n.id, { method: 'DELETE' })
    loadNotices()
  }

  return (
    <div className="mx-auto max-w-6xl space-y-5">
      <div>
        <h1 className="flex items-center gap-2 text-2xl font-bold text-slate-900">
          <LifeBuoy className="text-[#0284c7]" size={24} /> Soporte
          {unreadOpen > 0 && <span className="rounded-full bg-red-500 px-2.5 py-0.5 text-xs font-bold text-white">{unreadOpen} nuevo{unreadOpen > 1 ? 's' : ''}</span>}
        </h1>
        <p className="text-sm text-slate-500">Responde a quienes piden hablar con soporte desde la página principal y administra los avisos fijos del asistente.</p>

                <p className="min-w-0 flex-1 whitespace-pre-wrap break-words text-sm text-slate-700">{n.content}</p>

                <button onClick={() => delNotice(n)} title="Eliminar" className="rounded-lg p-1.5 text-slate-400 hover:bg-white hover:text-red-500"><Trash2 size={16} /></button>
              </li>
            ))}
          </ul>
        </section>
      )}
    </div>
  )
}

async function safeJson(res: Response): Promise<any> {
  const text = await res.text()
  if (!text) return {}
  try {
    return JSON.parse(text)
  } catch {
    return { error: `Respuesta no válida del servidor (HTTP ${res.status})` }
  }
}

function hm(d: string) {
  return new Date(d).toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' })

function dayLabel(d: string) {
  const x = new Date(d)
  const t = new Date()
  const y = new Date(Date.now() - 86400000)
  if (x.toDateString() === t.toDateString()) return 'Hoy'
  if (x.toDateString() === y.toDateString()) return 'Ayer'
  return x.toLocaleDateString('es-AR', { weekday: 'long', day: 'numeric', month: 'long' })
}
function untilText(s: Me['sanction']) {
  if (!s) return ''
  if (s.kind === 'BAN') return 'de forma permanente'
  if (!s.until) return ''
  return 'hasta el ' + new Date(s.until).toLocaleString('es-AR', { day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit' })
}

function fmtTime(d: string) {

function countdown(ms: number) {
  if (ms <= 0) return 'Ahora'
  const m = Math.floor(ms / 60000)
  const d = Math.floor(m / 1440)
  const h = Math.floor((m % 1440) / 60)
  if (d > 0) return `en ${d} d ${h} h`
  if (h > 0) return `en ${h} h ${m % 60} min`
  return `en ${m} min`
}
function ics(c: Call) {
  const z = (d: Date) => d.toISOString().replace(/[-:]/g, '').replace(/\.\d{3}/, '')
  const s = new Date(c.startsAt)
  const e = new Date(s.getTime() + c.durationMin * 60000)
  const esc = (t: string) => t.replace(/\\/g, '\\\\').replace(/;/g, '\\;').replace(/,/g, '\\,').replace(/\n/g, '\\n')
  const body = ['BEGIN:VCALENDAR', 'VERSION:2.0', 'PRODID:-//Invictus//Campus//ES', 'BEGIN:VEVENT', `UID:${c.id}@invictus`, `DTSTAMP:${z(new Date())}`, `DTSTART:${z(s)}`, `DTEND:${z(e)}`, `SUMMARY:${esc(c.title)}`, `DESCRIPTION:${esc((c.description || '') + '\nEl enlace se habilita en el campus 30 min antes.')}`, 'END:VEVENT', 'END:VCALENDAR'].join('\r\n')
  const a = document.createElement('a')
  a.href = URL.createObjectURL(new Blob([body], { type: 'text/calendar' }))
  a.download = 'invictus-llamada.ics'
  a.click()
  setTimeout(() => URL.revokeObjectURL(a.href), 2000)
}

}

// Recorta al centro y reduce a 256x256 para que pese poco
function resizeImage(file: File): Promise<string> {
  return new Promise((resolve, reject) => {
    const img = new Image()
    const url = URL.createObjectURL(file)
    img.onload = () => {
      const size = 256
      const canvas = document.createElement('canvas')
      canvas.width = size
      canvas.height = size
      const ctx = canvas.getContext('2d')
      if (!ctx) return reject(new Error('Tu navegador no puede procesar la imagen'))
      const s = Math.min(img.width, img.height)
      ctx.drawImage(img, (img.width - s) / 2, (img.height - s) / 2, s, s, 0, 0, size, size)
      URL.revokeObjectURL(url)
      resolve(canvas.toDataURL('image/jpeg', 0.85))
    }
    img.onerror = () => { URL.revokeObjectURL(url); reject(new Error('No se pudo leer la imagen. Usa JPG, PNG o WebP.')) }
    img.src = url
  })
}

import { authOptions } from '@/lib/auth'

import DashboardShell from '@/components/dashboard/DashboardShell'

export const dynamic = 'force-dynamic'

export default async function DashboardLayout({ children }: { children: ReactNode }) {
  const session: any = await getServerSession(authOptions)
  const id = session?.user?.id
  if (!id) redirect('/login')

  const user = await prismaDash.user.findUnique({
    where: { id },
    select: { name: true, email: true, role: true, avatarUrl: true, status: true, isBanned: true },
  })
  if (!user || user.isBanned || user.status !== 'APPROVED') redirect('/login')

  return (
    <DashboardShell user={{ name: user.name, email: user.email, role: user.role, avatarUrl: user.avatarUrl }}>
      {children}
    </DashboardShell>
  )
}
