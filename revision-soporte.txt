import { NextResponse } from 'next/server'
import { prismaDash } from '@/lib/prisma-dashboard'
import { cleanText } from '@/lib/chat-utils'
import { clientIpHash, newToken, short, EMAIL_RE } from '@/lib/support-utils'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

// Un visitante abre una conversación con soporte.
export async function POST(req: Request) {
  try {
    const body = await req.json().catch(() => ({}))
    const name = short(body?.name, 60)
    const email = short(body?.email, 120)
    const message = cleanText(body?.message)

    if (!name) return NextResponse.json({ error: 'Escribe tu nombre.' }, { status: 400 })
    if (email && !EMAIL_RE.test(email)) return NextResponse.json({ error: 'El correo no parece válido.' }, { status: 400 })
    if (!message) return NextResponse.json({ error: 'Escribe tu consulta.' }, { status: 400 })
    if (message.length > 1000) return NextResponse.json({ error: 'La consulta es demasiado larga (máximo 1000 caracteres).' }, { status: 400 })

    const ipHash = clientIpHash(req)
    const recent = await prismaDash.supportThread.count({
      where: { ipHash, createdAt: { gt: new Date(Date.now() - 3600 * 1000) } },
    })
    if (recent >= 3) {
      return NextResponse.json({ error: 'Ya abriste varias conversaciones. Espera un rato o continúa en la anterior.' }, { status: 429 })
    }

    const token = newToken()
    await prismaDash.supportThread.create({
      data: {
        token,
        name,
        email: email || null,
        ipHash,
        unreadAdmin: 1,
        messages: { create: { from: 'VISITOR', content: message } },
      },
    })
    return NextResponse.json({ token })
  } catch (err: any) {
    return NextResponse.json({ error: 'No pudimos enviar tu consulta. Inténtalo de nuevo.' }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import { prismaDash } from '@/lib/prisma-dashboard'
import { cleanText } from '@/lib/chat-utils'
import { TOKEN_RE } from '@/lib/support-utils'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

const NO_STORE = { 'Cache-Control': 'no-store' }

// El visitante consulta su conversación (el token secreto es su "llave").
export async function GET(_req: Request, { params }: { params: { token: string } }) {
  try {
    if (!TOKEN_RE.test(params.token)) return NextResponse.json({ error: 'No encontrada' }, { status: 404, headers: NO_STORE })
    const thread = await prismaDash.supportThread.findUnique({
      where: { token: params.token },
      include: { messages: { orderBy: { createdAt: 'asc' }, take: 200 } },
    })
    if (!thread) return NextResponse.json({ error: 'No encontrada' }, { status: 404, headers: NO_STORE })
    return NextResponse.json(
      {
        thread: { status: thread.status, name: thread.name },
        messages: thread.messages.map((m: any) => ({ id: m.id, from: m.from, content: m.content, createdAt: m.createdAt })),
      },
      { headers: NO_STORE }
    )
  } catch {
    return NextResponse.json({ error: 'Error interno' }, { status: 500, headers: NO_STORE })
  }
}

// El visitante envía otro mensaje.
export async function POST(req: Request, { params }: { params: { token: string } }) {
  try {
    if (!TOKEN_RE.test(params.token)) return NextResponse.json({ error: 'No encontrada' }, { status: 404 })
    const thread = await prismaDash.supportThread.findUnique({ where: { token: params.token } })
    if (!thread) return NextResponse.json({ error: 'No encontrada' }, { status: 404 })
    if (thread.status === 'CLOSED') return NextResponse.json({ error: 'Esta conversación fue cerrada.' }, { status: 409 })

    const body = await req.json().catch(() => ({}))
    const content = cleanText(body?.message)
    if (!content) return NextResponse.json({ error: 'Escribe un mensaje.' }, { status: 400 })
    if (content.length > 1000) return NextResponse.json({ error: 'El mensaje es demasiado largo.' }, { status: 400 })

    const lastStaff = await prismaDash.supportMessage.findFirst({
      where: { threadId: thread.id, from: 'STAFF' },
      orderBy: { createdAt: 'desc' },
      select: { createdAt: true },
    })
    const pending = await prismaDash.supportMessage.findMany({
      where: { threadId: thread.id, from: 'VISITOR', ...(lastStaff ? { createdAt: { gt: lastStaff.createdAt } } : {}) },
      orderBy: { createdAt: 'desc' },
      take: 10,
      select: { createdAt: true },
    })
    if (pending.length && Date.now() - pending[0].createdAt.getTime() < 2000) {
      return NextResponse.json({ error: 'Vas muy rápido. Espera un momento.' }, { status: 429 })
    }
    if (pending.length >= 8) {
      return NextResponse.json({ error: 'Ya enviaste varios mensajes. Espera la respuesta del equipo.' }, { status: 429 })
    }

    const message = await prismaDash.supportMessage.create({ data: { threadId: thread.id, from: 'VISITOR', content } })
    await prismaDash.supportThread.update({
      where: { id: thread.id },
      data: { unreadAdmin: { increment: 1 }, lastMessageAt: new Date() },
    })
    return NextResponse.json({ message: { id: message.id, from: message.from, content: message.content, createdAt: message.createdAt } })
  } catch {
    return NextResponse.json({ error: 'No se pudo enviar.' }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import { prismaDash } from '@/lib/prisma-dashboard'
import { getStaff } from '@/lib/chat-utils'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

// Lista de conversaciones de soporte (las más recientes primero).
export async function GET() {
  try {
    const admin = await getStaff()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const rows = await prismaDash.supportThread.findMany({
      orderBy: { lastMessageAt: 'desc' },
      take: 100,
      include: { messages: { orderBy: { createdAt: 'desc' }, take: 1 } },
    })
    const threads = rows.map((t: any) => ({
      id: t.id,
      name: t.name,
      email: t.email,
      status: t.status,
      unreadAdmin: t.unreadAdmin,
      lastMessageAt: t.lastMessageAt,
      createdAt: t.createdAt,
      last: t.messages[0] ? { from: t.messages[0].from, content: t.messages[0].content } : null,
    }))
    const unreadTotal = threads.filter((t: any) => t.status === 'OPEN').reduce((n: number, t: any) => n + (t.unreadAdmin > 0 ? 1 : 0), 0)
    return NextResponse.json({ threads, unreadTotal }, { headers: { 'Cache-Control': 'no-store' } })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import { prismaDash } from '@/lib/prisma-dashboard'
import { getAdmin, cleanText } from '@/lib/chat-utils'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

// Conversación completa; al abrirla se marca como leída.
export async function GET(_req: Request, { params }: { params: { id: string } }) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    const thread = await prismaDash.supportThread.findUnique({
      where: { id: params.id },
      include: { messages: { orderBy: { createdAt: 'asc' }, take: 300 } },
    })
    if (!thread) return NextResponse.json({ error: 'No existe' }, { status: 404 })
    if (thread.unreadAdmin > 0) await prismaDash.supportThread.update({ where: { id: thread.id }, data: { unreadAdmin: 0 } })
    return NextResponse.json(
      {
        thread: { id: thread.id, name: thread.name, email: thread.email, status: thread.status, createdAt: thread.createdAt },
        messages: thread.messages.map((m: any) => ({ id: m.id, from: m.from, content: m.content, createdAt: m.createdAt })),
      },
      { headers: { 'Cache-Control': 'no-store' } }
    )
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

// Responder al visitante.
export async function POST(req: Request, { params }: { params: { id: string } }) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    const body = await req.json().catch(() => ({}))
    const content = cleanText(body?.content)
    if (!content) return NextResponse.json({ error: 'Escribe una respuesta.' }, { status: 400 })
    if (content.length > 2000) return NextResponse.json({ error: 'La respuesta es demasiado larga.' }, { status: 400 })

    const thread = await prismaDash.supportThread.findUnique({ where: { id: params.id }, select: { id: true } })
    if (!thread) return NextResponse.json({ error: 'No existe' }, { status: 404 })

    const message = await prismaDash.supportMessage.create({ data: { threadId: thread.id, from: 'STAFF', content } })
    await prismaDash.supportThread.update({ where: { id: thread.id }, data: { status: 'OPEN', unreadAdmin: 0, lastMessageAt: new Date() } })
    return NextResponse.json({ message: { id: message.id, from: message.from, content: message.content, createdAt: message.createdAt } })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

// Cerrar o reabrir.
export async function PATCH(req: Request, { params }: { params: { id: string } }) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    const { status } = await req.json().catch(() => ({}))
    if (status !== 'OPEN' && status !== 'CLOSED') return NextResponse.json({ error: 'Estado inválido.' }, { status: 400 })
    await prismaDash.supportThread.update({ where: { id: params.id }, data: { status } })
    return NextResponse.json({ success: true })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

export async function DELETE(_req: Request, { params }: { params: { id: string } }) {
  try {
    const admin = await getAdmin()
    if (!admin) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    await prismaDash.supportThread.deleteMany({ where: { id: params.id } })
    return NextResponse.json({ success: true })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

  id: string
  name: string
  email: string | null
  status: 'OPEN' | 'CLOSED'
  unreadAdmin: number
  lastMessageAt: string
  last: { from: string; content: string } | null
}
interface TMsg {
  id: string
  from: 'VISITOR' | 'STAFF'
  content: string
  createdAt: string
}
interface Notice {
  id: string
  content: string
  isActive: boolean
}

function ago(d: string) {
  const m = Math.floor((Date.now() - new Date(d).getTime()) / 60000)
  if (m < 1) return 'ahora'
  if (m < 60) return `${m} min`
  if (m < 1440) return `${Math.floor(m / 60)} h`
  return `${Math.floor(m / 1440)} d`
}
function hm(d: string) {
  return new Date(d).toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' })
}

  const [sel, setSel] = useState<string | null>(null)
  const [conv, setConv] = useState<{ thread: any; messages: TMsg[] } | null>(null)
  const [text, setText] = useState('')
  const [busy, setBusy] = useState(false)

    if (el && stick.current) el.scrollTop = el.scrollHeight

  const list = threads.filter((t) => t.status === filter)

import { useEffect, useRef } from 'react'
import { LANDING_CSS } from './landing-css'
import { LANDING_HTML } from './landing-html'
import { LANDING_JS } from './landing-js'

// 'youtube' -> usa tu video de YouTube (recomendado)
// 'file'    -> usa el archivo /public/landing/video.mp4 y /public/landing/poster.jpg
const VIDEO_MODE = 'youtube' as 'youtube' | 'file'

const POSTER =
  VIDEO_MODE === 'file'
    ? '/landing/poster.jpg'
    : 'https://i.ytimg.com/vi/H-t8Sk2FhA0/maxresdefault.jpg'

const HTML = LANDING_HTML.split('__POSTER__').join(POSTER)

export default function Landing() {
  const ref = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const el = ref.current
    if (!el) return
    const root = document.documentElement
    const prevScroll = root.style.scrollBehavior
    root.style.scrollBehavior = 'smooth'

    const init = new Function('R', 'MODE', LANDING_JS) as (
      R: HTMLElement,
      MODE: string
    ) => () => void
    const cleanup = init(el, VIDEO_MODE)

    return () => {
      if (typeof cleanup === 'function') cleanup()
      root.style.scrollBehavior = prevScroll
    }
  }, [])

  return (
    <>
      <style dangerouslySetInnerHTML={{ __html: LANDING_CSS }} />
      <div
        id="invl-root"
        ref={ref}
        className="invl"
        suppressHydrationWarning
        dangerouslySetInnerHTML={{ __html: HTML }}

'use client'

import { useCallback, useEffect, useRef, useState } from 'react'
import { answerFor, BOT_CONTACT, GREETING, type BotAnswer } from '@/lib/faq-bot'

interface Msg {
  id: number
  from: 'bot' | 'me'
  text: string
  actions?: { label: string; href: string }[]
  follow?: string[]
}
interface SMsg {
  id: string
  from: 'VISITOR' | 'STAFF'
  content: string
  createdAt: string
}

const KEY = 'inv-bot-v1'
const TKEY = 'inv-support-token'
const SEENKEY = 'inv-support-seen'

function hm(d: string) {
  return new Date(d).toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' })
}

export default function LandingBot() {
  const [open, setOpen] = useState(false)
  const [view, setView] = useState<'bot' | 'support'>('bot')
  const [msgs, setMsgs] = useState<Msg[]>([])
  const [input, setInput] = useState('')
  const [typing, setTyping] = useState(false)
  const [unread, setUnread] = useState(true)
  const [notices, setNotices] = useState<{ id: string; content: string }[]>([])
  const end = useRef<HTMLDivElement>(null)
  const idRef = useRef(1)
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null)
  const noticesLoaded = useRef(false)

  // soporte
  const [token, setToken] = useState<string | null>(null)
  const [sMsgs, setSMsgs] = useState<SMsg[]>([])
  const [sClosed, setSClosed] = useState(false)
  const [sInput, setSInput] = useState('')
  const [sErr, setSErr] = useState('')
  const [sBusy, setSBusy] = useState(false)
  const [form, setForm] = useState({ name: '', email: '', message: '' })
  const [replyDot, setReplyDot] = useState(false)
  const seenStaff = useRef(0)
  const sEnd = useRef<HTMLDivElement>(null)

  useEffect(() => {
    let restored: Msg[] | null = null
    try {
      const raw = sessionStorage.getItem(KEY)
      if (raw) restored = JSON.parse(raw)
    } catch {}
    if (restored && restored.length) {
      setMsgs(restored)
      idRef.current = restored.reduce((m, x) => Math.max(m, x.id), 0) + 1
    } else {
      setMsgs([{ id: 0, from: 'bot', text: GREETING.text, follow: GREETING.follow }])
    }
    try {
      const t = localStorage.getItem(TKEY)
      if (t && /^[a-f0-9]{48}$/.test(t)) setToken(t)
      seenStaff.current = parseInt(localStorage.getItem(SEENKEY) || '0', 10) || 0
    } catch {}
    return () => {
      if (timer.current) clearTimeout(timer.current)
    }
  }, [])

  useEffect(() => {
    try {
      if (msgs.length) sessionStorage.setItem(KEY, JSON.stringify(msgs.slice(-40)))
    } catch {}
    end.current?.scrollIntoView({ behavior: 'smooth', block: 'end' })
  }, [msgs, typing, open, view])

  useEffect(() => {
    sEnd.current?.scrollIntoView({ behavior: 'smooth', block: 'end' })
  }, [sMsgs, view, open])

  // Avisos fijos del equipo
  useEffect(() => {
    if (!open || noticesLoaded.current) return
    noticesLoaded.current = true
    fetch('/api/public/bot-notices')
      .then((r) => r.json())
      .then((d) => setNotices(d.notices || []))
      .catch(() => {})
  }, [open])

  const loadSupport = useCallback(async () => {
    if (!token) return
    try {
      const r = await fetch('/api/public/support/' + token, { cache: 'no-store' })
      if (r.status === 404) {
        try { localStorage.removeItem(TKEY); localStorage.removeItem(SEENKEY) } catch {}
        seenStaff.current = 0
        setToken(null)
        setSMsgs([])
        return
      }
      const d = await r.json()
      if (!r.ok) return
      setSMsgs(d.messages)
      setSClosed(d.thread.status === 'CLOSED')
      const staff = d.messages.filter((m: SMsg) => m.from === 'STAFF').length
      if (open && view === 'support') {
        seenStaff.current = staff
        try { localStorage.setItem(SEENKEY, String(staff)) } catch {}
        setReplyDot(false)
      } else if (staff > seenStaff.current) {
        setReplyDot(true)
      }
    } catch {}
  }, [token, open, view])

  useEffect(() => {
    if (!token) return
    loadSupport()
    const active = open && view === 'support'
    const t = setInterval(() => !document.hidden && loadSupport(), active ? 4000 : 20000)
    return () => clearInterval(t)
  }, [token, open, view, loadSupport])

  function push(from: 'bot' | 'me', a: Partial<BotAnswer> & { text: string }) {
    setMsgs((m) => [...m, { id: idRef.current++, from, text: a.text, actions: a.actions, follow: a.follow }])
  }

  function ask(q: string) {
    const text = q.trim().slice(0, 200)
    if (!text || typing) return
    push('me', { text })
    setInput('')
    setTyping(true)
    const ans = answerFor(text)
    timer.current = setTimeout(() => {
      setTyping(false)
      push('bot', ans)
    }, 450 + Math.min(ans.text.length * 4, 700))
  }

  function openSupport() {

    setView('support')

    } catch {

    } finally {
      setSBusy(false)
    }
  }

      else {
        setSInput('')
        setSMsgs((m) => [...m, d.message])
      }
    } catch {
      setSErr('Sin conexión. Inténtalo de nuevo.')
    } finally {
      setSBusy(false)
    }
  }

  }

          </svg>
        ) : (
          <svg viewBox="0 0 24 24" width="26" height="26" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
            <path d="M21 12a8 8 0 0 1-11.6 7.1L4 20l1-4.6A8 8 0 1 1 21 12z" />
            <path d="M8.5 11.5h.01M12 11.5h.01M15.5 11.5h.01" strokeWidth="2.6" />

        )}
        {showDot && <span className={'ib-dot' + (replyDot ? ' ib-red' : '')} />}

// ---------- Motor de coincidencia ----------

function normalize(s: string): string {
  return s
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9\s]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim()
}

function stem(w: string): string {
  return w.length > 4 && w.endsWith('s') ? w.slice(0, -1) : w
}

function tokens(s: string): string[] {
  const n = normalize(s)
  return n ? n.split(' ').map(stem) : []
}

function lev(a: string, b: string): number {
  if (a === b) return 0
  if (Math.abs(a.length - b.length) > 1) return 2
  const dp: number[] = Array.from({ length: b.length + 1 }, (_, i) => i)
  for (let i = 1; i <= a.length; i++) {
    let prev = dp[0]
    dp[0] = i
    for (let j = 1; j <= b.length; j++) {
      const tmp = dp[j]
      dp[j] = Math.min(dp[j] + 1, dp[j - 1] + 1, prev + (a[i - 1] === b[j - 1] ? 0 : 1))
      prev = tmp
    }
  }
  return dp[b.length]
}

function sameWord(a: string, b: string): boolean {
  if (a === b) return true
  return a.length >= 5 && b.length >= 5 && lev(a, b) <= 1
}

function hasPhrase(msg: string[], phrase: string[]): boolean {
  if (!phrase.length) return false
  for (let i = 0; i + phrase.length <= msg.length; i++) {
    let ok = true
    for (let j = 0; j < phrase.length; j++) {
      if (!sameWord(msg[i + j], phrase[j])) {
        ok = false
        break
      }
    }
    if (ok) return true
  }
  return false
}

export function answerFor(question: string): BotAnswer {
  const msg = tokens(question)
  if (!msg.length) return FALLBACK

  let best: Entry | null = null
  let bestScore = 0
  for (const e of ENTRIES) {
    let score = 0
    for (const p of e.strong) if (hasPhrase(msg, tokens(p))) score += 3 + (tokens(p).length > 1 ? 1 : 0)
    for (const w of e.weak || []) if (hasPhrase(msg, tokens(w))) score += 1
    if (score > bestScore) {
      best = e
      bestScore = score
    }
  }
  // Un saludo no debe ganarle a una pregunta real.
  if (best && best.id === 'saludo' && msg.length > 4) return answerFor(msg.filter((w) => !['hola', 'buena', 'buenas', 'hey'].includes(w)).join(' '))
  return best && (bestScore >= 2 || (bestScore >= 1 && msg.length <= 2)) ? best.answer : FALLBACK
}

import crypto from 'crypto'

export const TOKEN_RE = /^[a-f0-9]{48}$/
export const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/

export const newToken = () => crypto.randomBytes(24).toString('hex')

// Huella anónima del visitante (no se guarda la IP real) para limitar abusos.
export function clientIpHash(req: Request): string {
  const h = req.headers
  const ip =
    h.get('x-nf-client-connection-ip') ||
    h.get('x-forwarded-for')?.split(',')[0]?.trim() ||
    h.get('x-real-ip') ||
    'unknown'
  return crypto
    .createHash('sha256')
    .update(ip + (process.env.NEXTAUTH_SECRET || 'invictus'))
    .digest('hex')
    .slice(0, 32)
}

export function short(input: unknown, max: number): string {
  if (typeof input !== 'string') return ''
  return input.replace(/[\u0000-\u001F\u007F]/g, ' ').replace(/\s+/g, ' ').trim().slice(0, max)
}
