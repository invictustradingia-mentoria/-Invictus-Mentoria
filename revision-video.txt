'use client'

import { useCallback, useEffect, useRef, useState } from 'react'
import { Play, Pause, Volume2, VolumeX, Maximize, Minimize, Loader2, AlertCircle } from 'lucide-react'

declare global {
  interface Window { YT?: any; onYouTubeIframeAPIReady?: () => void }
}

type Source =
  | { type: 'youtube'; id: string; watermark: string }
  | { type: 'file'; url: string; watermark: string }
  | { type: 'embed'; src: string; watermark: string }

let ytPromise: Promise<any> | null = null
function loadYT(): Promise<any> {
  if (typeof window === 'undefined') return Promise.reject()
  if (window.YT?.Player) return Promise.resolve(window.YT)
  if (!ytPromise) {
    ytPromise = new Promise((resolve) => {
      const prev = window.onYouTubeIframeAPIReady
      window.onYouTubeIframeAPIReady = () => { prev?.(); resolve(window.YT) }
      const s = document.createElement('script')
      s.src = 'https://www.youtube.com/iframe_api'
      document.head.appendChild(s)
    })
  }
  return ytPromise
}

const fmt = (s: number) => {
  if (!isFinite(s) || s < 0) s = 0
  const m = Math.floor(s / 60)
  return `${m}:${String(Math.floor(s % 60)).padStart(2, '0')}`
}

const RATES = [0.75, 1, 1.25, 1.5, 2]

export default function SecurePlayer({ lessonId, title }: { lessonId: string; title: string }) {
  const boxRef = useRef<HTMLDivElement>(null)
  const ytHost = useRef<HTMLDivElement>(null)
  const videoRef = useRef<HTMLVideoElement>(null)
  const ytPlayer = useRef<any>(null)
  const tick = useRef<any>(null)

  const [src, setSrc] = useState<Source | null>(null)
  const [error, setError] = useState<string | null>(null)
  const [ready, setReady] = useState(false)
  const [started, setStarted] = useState(false)
  const [playing, setPlaying] = useState(false)
  const [cur, setCur] = useState(0)
  const [dur, setDur] = useState(0)
  const [muted, setMuted] = useState(false)
  const [rate, setRate] = useState(1)
  const [full, setFull] = useState(false)
  const [wm, setWm] = useState({ top: 12, left: 12 })

  // 1. Pedir el video al servidor (valida sesión y nivel)
  useEffect(() => {
    let alive = true
    setSrc(null); setError(null); setReady(false); setStarted(false); setPlaying(false); setCur(0); setDur(0)
    fetch(`/api/student/video?lessonId=${encodeURIComponent(lessonId)}`, { cache: 'no-store' })
      .then(async (r) => {
        const d = await r.json().catch(() => ({}))
        if (!alive) return
        if (!r.ok) setError(d.error || `No se pudo cargar el video (HTTP ${r.status})`)
        else setSrc(d)
      })
      .catch(() => alive && setError('No se pudo conectar con el servidor.'))
    return () => { alive = false }
  }, [lessonId])

  // 2. YouTube con controles propios
  useEffect(() => {
    if (!src || src.type !== 'youtube') return
    let destroyed = false
    loadYT().then((YT) => {
      if (destroyed || !ytHost.current) return
      const mount = document.createElement('div')
      ytHost.current.innerHTML = ''
      ytHost.current.appendChild(mount)
      ytPlayer.current = new YT.Player(mount, {
        videoId: src.id,
        host: 'https://www.youtube-nocookie.com',
        width: '100%',
        height: '100%',
        playerVars: {
          controls: 0, disablekb: 1, fs: 0, modestbranding: 1, rel: 0, iv_load_policy: 3,
          playsinline: 1, cc_load_policy: 0, origin: window.location.origin,
        },
        events: {
          onReady: (e: any) => { setDur(e.target.getDuration()); setReady(true) },
          onStateChange: (e: any) => {
            setPlaying(e.data === 1)
            if (e.data === 1) setStarted(true)
            if (e.data === 1 || e.data === 2) setDur(e.target.getDuration())
          },
          onError: (e: any) => {
            const map: Record<number, string> = {
              2: 'El enlace del video no es válido.',
              5: 'Error de reproducción en tu navegador.',
              100: 'El video fue eliminado o está privado en su origen.',
              101: 'El dueño del video no permite reproducirlo en otras páginas.',
              150: 'El dueño del video no permite reproducirlo en otras páginas.',
              153: 'El video no permite reproducirse aquí (configuración de inserción).',
            }
            setError(map[e.data] || 'No se pudo reproducir el video.')
          },
        },
      })
      tick.current = setInterval(() => {
        const p = ytPlayer.current
        if (p?.getCurrentTime) setCur(p.getCurrentTime())
      }, 400)
    }).catch(() => setError('No se pudo cargar el reproductor.'))

    return () => {
      destroyed = true
      clearInterval(tick.current)
      try { ytPlayer.current?.destroy() } catch {}
      ytPlayer.current = null
    }
  }, [src])

  // Marca de agua: cambia de lugar
  useEffect(() => {
    const t = setInterval(() => setWm({ top: 8 + Math.random() * 72, left: 4 + Math.random() * 58 }), 7000)
    return () => clearInterval(t)
  }, [])

  // Pausar al cambiar de pestaña
  useEffect(() => {
    const onHide = () => { if (document.hidden) pause() }
    document.addEventListener('visibilitychange', onHide)
    return () => document.removeEventListener('visibilitychange', onHide)
  })

  useEffect(() => {
    const onFs = () => setFull(!!document.fullscreenElement)
    document.addEventListener('fullscreenchange', onFs)
    return () => document.removeEventListener('fullscreenchange', onFs)
  }, [])

  // Intento de bloquear capturas por teclado (disuasivo)
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'PrintScreen') {
        pause()
        try { navigator.clipboard.writeText('') } catch {}
      }
    }
    window.addEventListener('keyup', onKey)
    return () => window.removeEventListener('keyup', onKey)
  })

  const play = useCallback(() => {
    if (src?.type === 'youtube') ytPlayer.current?.playVideo?.()
    else videoRef.current?.play()
  }, [src])

  function pause() {
    if (src?.type === 'youtube') ytPlayer.current?.pauseVideo?.()
    else videoRef.current?.pause()
  }

  const toggle = () => (playing ? pause() : play())

  const seek = (t: number) => {
    setCur(t)
    if (src?.type === 'youtube') ytPlayer.current?.seekTo?.(t, true)
    else if (videoRef.current) videoRef.current.currentTime = t
  }

  const toggleMute = () => {
    const next = !muted
    setMuted(next)
    if (src?.type === 'youtube') next ? ytPlayer.current?.mute?.() : ytPlayer.current?.unMute?.()
    else if (videoRef.current) videoRef.current.muted = next
  }

  const cycleRate = () => {
    const next = RATES[(RATES.indexOf(rate) + 1) % RATES.length]
    setRate(next)
    if (src?.type === 'youtube') ytPlayer.current?.setPlaybackRate?.(next)
    else if (videoRef.current) videoRef.current.playbackRate = next
  }

  const toggleFull = () => {
    if (document.fullscreenElement) document.exitFullscreen()
    else boxRef.current?.requestFullscreen?.()
  }

  const watermark = src?.watermark || ''

  return (
    <div
      ref={boxRef}
      onContextMenu={(e) => e.preventDefault()}
      className="relative aspect-video w-full select-none overflow-hidden rounded-2xl bg-black"
    >
      {error ? (
        <div className="flex h-full flex-col items-center justify-center gap-2 p-6 text-center text-white">
          <AlertCircle size={30} className="text-rose-400" />
          <p className="text-sm font-bold">{error}</p>
        </div>
      ) : !src ? (
        <div className="flex h-full items-center justify-center text-white/70"><Loader2 className="animate-spin" /></div>
      ) : src.type === 'embed' ? (
        <iframe src={src.src} title={title} className="h-full w-full" allow="autoplay; fullscreen; picture-in-picture" allowFullScreen />
      ) : (
        <>
          {src.type === 'youtube' ? (
            <div ref={ytHost} className="absolute inset-0 [&_iframe]:h-full [&_iframe]:w-full" />
          ) : (
            <video
              ref={videoRef}
              src={src.url}
              className="absolute inset-0 h-full w-full"
              controlsList="nodownload noplaybackrate noremoteplayback"
              disablePictureInPicture
              disableRemotePlayback
              playsInline
              preload="metadata"
              onLoadedMetadata={(e) => { setDur(e.currentTarget.duration); setReady(true) }}
              onTimeUpdate={(e) => setCur(e.currentTarget.currentTime)}
              onPlay={() => { setPlaying(true); setStarted(true) }}
              onPause={() => setPlaying(false)}
              onError={() => setError('No se pudo cargar el video. Es posible que el enlace haya vencido: recarga la página.')}
            />
          )}

          {/* Capa que bloquea cualquier clic sobre el reproductor original */}
          <div className="absolute inset-0 z-10" onClick={toggle} onDoubleClick={toggleFull} />

          {/* Portada: tapa el título y las sugerencias de YouTube mientras no se reproduce */}
          {!playing && (
            <div className="absolute inset-0 z-20 flex cursor-pointer flex-col items-center justify-center bg-gradient-to-br from-slate-900 to-slate-700 text-white" onClick={toggle}>
              {!ready ? (
                <Loader2 className="animate-spin" />
              ) : (
                <>
                  <span className="flex h-20 w-20 items-center justify-center rounded-full bg-sky-500 shadow-xl shadow-sky-500/40 transition-transform hover:scale-110">
                    <Play size={34} fill="white" className="ml-1" />
                  </span>
                  <p className="mt-4 max-w-md px-6 text-center text-sm font-bold">{started ? 'En pausa' : title}</p>
                </>
              )}
            </div>
          )}

          {/* Marca de agua con el correo del alumno */}
          <div
            className="pointer-events-none absolute z-30 whitespace-nowrap text-[11px] font-bold text-white/30 transition-all duration-[2000ms]"
            style={{ top: `${wm.top}%`, left: `${wm.left}%`, textShadow: '0 0 4px rgba(0,0,0,.6)' }}
          >
            {watermark}
          </div>

          {/* Controles propios */}
          <div className="absolute inset-x-0 bottom-0 z-40 bg-gradient-to-t from-black/85 to-transparent px-3 pb-2 pt-8" onClick={(e) => e.stopPropagation()}>
            <input
              type="range" min={0} max={dur || 0} step={0.5} value={Math.min(cur, dur || 0)}
              onChange={(e) => seek(Number(e.target.value))}
              aria-label="Progreso"
              className="h-1 w-full cursor-pointer accent-sky-400"
            />
            <div className="mt-1.5 flex items-center gap-3 text-white">
              <button type="button" onClick={toggle} aria-label={playing ? 'Pausar' : 'Reproducir'} className="cursor-pointer">
                {playing ? <Pause size={20} fill="white" /> : <Play size={20} fill="white" />}
              </button>
              <span className="text-[11px] font-bold tabular-nums">{fmt(cur)} / {fmt(dur)}</span>
              <div className="ml-auto flex items-center gap-3">
                <button type="button" onClick={cycleRate} className="rounded-md bg-white/15 px-2 py-0.5 text-[11px] font-black cursor-pointer">{rate}x</button>
                <button type="button" onClick={toggleMute} aria-label="Sonido" className="cursor-pointer">
                  {muted ? <VolumeX size={19} /> : <Volume2 size={19} />}
                </button>
                <button type="button" onClick={toggleFull} aria-label="Pantalla completa" className="cursor-pointer">
                  {full ? <Minimize size={19} /> : <Maximize size={19} />}
                </button>
              </div>
            </div>
          </div>
        </>
      )}
    </div>
  )
}

import { NextResponse } from 'next/server'
import { createClient } from '@supabase/supabase-js'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prismaDash } from '@/lib/prisma-dashboard'
import { visibleLevels } from '@/lib/access'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

function youtubeId(url: string): string | null {
  try {
    const u = new URL(url)
    const host = u.hostname.replace(/^www\./, '')
    if (host === 'youtu.be') return u.pathname.split('/').filter(Boolean)[0] || null
    if (host === 'youtube.com' || host === 'm.youtube.com' || host === 'youtube-nocookie.com') {
      if (u.pathname === '/watch') return u.searchParams.get('v')
      const parts = u.pathname.split('/')
      if (['embed', 'shorts', 'live'].includes(parts[1])) return parts[2] || null
    }
  } catch {}
  return null
}

function vimeoEmbed(url: string): string | null {
  try {
    const u = new URL(url)
    if (u.hostname.replace(/^www\./, '') !== 'vimeo.com') return null
    const id = u.pathname.split('/').filter(Boolean)[0]
    return id && /^\d+$/.test(id) ? `https://player.vimeo.com/video/${id}?title=0&byline=0&portrait=0` : null
  } catch {
    return null
  }
}

export async function GET(req: Request) {
  try {
    const session: any = await getServerSession(authOptions)
    if (!session?.user?.id) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const lessonId = new URL(req.url).searchParams.get('lessonId') || ''
    if (!lessonId) return NextResponse.json({ error: 'Falta la lección' }, { status: 400 })

    const lesson = await prismaDash.lesson.findUnique({ where: { id: lessonId }, include: { module: true } })
    if (!lesson || !lesson.isPublished || !lesson.module.isPublished) {
      return NextResponse.json({ error: 'La lección no existe' }, { status: 404 })
    }

    const levels = visibleLevels(session.user.role)
    if (levels && !levels.includes(lesson.allowedRole)) {
      return NextResponse.json({ error: 'No tienes acceso a esta lección' }, { status: 403 })
    }

    const watermark = session.user.email || session.user.name || 'Alumno'
    const url = lesson.videoUrl

    const yt = youtubeId(url)
    if (yt) return NextResponse.json({ type: 'youtube', id: yt, watermark })

    const vm = vimeoEmbed(url)
    if (vm) return NextResponse.json({ type: 'embed', src: vm, watermark })

    // Video subido a Supabase: enlace firmado de corta duración
    const marker = '/storage/v1/object/public/videos/'
    const idx = url.indexOf(marker)
    if (idx !== -1) {
      const path = decodeURIComponent(url.slice(idx + marker.length).split('?')[0])
      const supabaseUrl = (process.env.SUPABASE_URL || process.env.NEXT_PUBLIC_SUPABASE_URL || '').trim()
      const key = (process.env.SUPABASE_SERVICE_ROLE_KEY || '').trim()
      if (supabaseUrl && key) {
        const supabase = createClient(supabaseUrl, key, { auth: { persistSession: false } })
        const { data } = await supabase.storage.from('videos').createSignedUrl(path, 3600)
        if (data?.signedUrl) return NextResponse.json({ type: 'file', url: data.signedUrl, watermark })
      }
    }

    return NextResponse.json({ type: 'file', url, watermark })
  } catch (err: any) {
    console.error('GET student/video:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prismaDash } from '@/lib/prisma-dashboard'
import { visibleLevels } from '@/lib/access'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

export async function GET() {
  try {
    const session: any = await getServerSession(authOptions)
    const userId = session?.user?.id
    if (!userId) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const role: string = session.user.role ?? 'CLUB'
    const levels = visibleLevels(role)

    const modules = await prismaDash.module.findMany({
      where: { isPublished: true },
      orderBy: [{ order: 'asc' }, { createdAt: 'asc' }],
      include: {
        lessons: {
          where: { isPublished: true, ...(levels ? { allowedRole: { in: levels } } : {}) },
          orderBy: [{ order: 'asc' }, { createdAt: 'asc' }],
          // videoUrl NO se envía al navegador: el reproductor lo pide por /api/student/video
          select: { id: true, title: true, description: true, duration: true, order: true, allowedRole: true },
        },
      },
    })

    const visible = modules
      .filter((m) => m.lessons.length > 0)
      .map((m) => ({ id: m.id, title: m.title, description: m.description, lessons: m.lessons }))

    const ids = visible.flatMap((m) => m.lessons.map((l) => l.id))
    const rows = ids.length
      ? await prismaDash.lessonProgress.findMany({ where: { userId, lessonId: { in: ids } } })
      : []

    const progress: Record<string, { completed: boolean }> = {}
    rows.forEach((r) => { progress[r.lessonId] = { completed: r.completed } })

    return NextResponse.json({ role, modules: visible, progress })
  } catch (err: any) {
    console.error('GET student/courses:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

import { ACCESS_LEVELS } from '@/lib/access'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

export async function POST(req: Request) {
  try {
    const session: any = await getServerSession(authOptions)
    if (!session?.user || session.user.role !== 'ADMIN') {
      return NextResponse.json({ error: 'No autorizado' }, { status: 401 })
    }

    const body = await req.json()
    const title = typeof body.title === 'string' ? body.title.trim() : ''
    const videoUrl = typeof body.videoUrl === 'string' ? body.videoUrl.trim() : ''
    const moduleId = typeof body.moduleId === 'string' ? body.moduleId : ''
    const description = typeof body.description === 'string' ? body.description.trim() : ''
    const allowedRole = typeof body.allowedRole === 'string' ? body.allowedRole : 'ALL'

    if (!title) return NextResponse.json({ error: 'Falta el título de la clase' }, { status: 400 })
    if (!/^https?:\/\//i.test(videoUrl)) return NextResponse.json({ error: 'El enlace del video debe empezar con http:// o https://' }, { status: 400 })
    if (!ACCESS_LEVELS.includes(allowedRole)) return NextResponse.json({ error: 'Nivel de acceso no válido' }, { status: 400 })

    const mod = await prismaDash.module.findUnique({ where: { id: moduleId } })
    if (!mod) return NextResponse.json({ error: 'El módulo seleccionado no existe' }, { status: 400 })

    const count = await prismaDash.lesson.count({ where: { moduleId } })
    const d = parseInt(String(body.duration), 10)

    const lesson = await prismaDash.lesson.create({
      data: {
        title,
        description: description || null,
        videoUrl,
        duration: Number.isFinite(d) && d > 0 ? d : null,
        allowedRole,
        moduleId,
        order: count + 1,
        isPublished: true,
      },
    })

    return NextResponse.json({ success: true, lesson })
  } catch (err: any) {
    console.error('POST lessons:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

async function isAdmin() {

export async function GET(_req: Request, { params }: { params: { id: string } }) {

    const lesson = await prismaDash.lesson.findUnique({ where: { id: params.id } })

    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }

export async function PATCH(req: Request, { params }: { params: { id: string } }) {
  try {
    if (!(await isAdmin())) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const body = await req.json()
    const data: Record<string, any> = {}

    if (typeof body.title === 'string') {
      if (!body.title.trim()) return NextResponse.json({ error: 'El título no puede estar vacío' }, { status: 400 })
      data.title = body.title.trim()
    }
    if (typeof body.description === 'string') data.description = body.description.trim() || null
    if (typeof body.videoUrl === 'string') {
      if (!body.videoUrl.trim()) return NextResponse.json({ error: 'El enlace del video no puede estar vacío' }, { status: 400 })
      data.videoUrl = body.videoUrl.trim()
    }
    if (body.duration !== undefined) {
      const d = parseInt(String(body.duration), 10)
      data.duration = Number.isFinite(d) && d > 0 ? d : null
    }
    if (body.order !== undefined) {
      const o = parseInt(String(body.order), 10)
      if (Number.isFinite(o)) data.order = o
    }
    if (typeof body.isPublished === 'boolean') data.isPublished = body.isPublished
    if (typeof body.allowedRole === 'string') {
      if (!ROLES.includes(body.allowedRole)) return NextResponse.json({ error: 'Rol no válido' }, { status: 400 })
      data.allowedRole = body.allowedRole
    }
    if (typeof body.moduleId === 'string' && body.moduleId) {
      const mod = await prismaDash.module.findUnique({ where: { id: body.moduleId } })
      if (!mod) return NextResponse.json({ error: 'El módulo destino no existe' }, { status: 400 })
      data.moduleId = body.moduleId
    }

    const lesson = await prismaDash.lesson.update({ where: { id: params.id }, data })
    return NextResponse.json({ success: true, lesson })
  } catch (err: any) {
    console.error('PATCH lesson:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export const dynamic = 'force-dynamic'

export async function POST(req: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    if (!session || !session.user) {
      return NextResponse.json({ error: 'No autenticado' }, { status: 401 })
    }

    const userId = (session.user as any).id
    if (!userId) {
      return NextResponse.json({ error: 'ID de usuario no encontrado' }, { status: 400 })
    }

    const body = await req.json()
    const { action, page, lessonId, lessonTitle, metadata } = body

    const forwardedFor = req.headers.get('x-forwarded-for')
    const realIp = req.headers.get('x-real-ip')
    const clientIp = forwardedFor ? forwardedFor.split(',')[0].trim() : (realIp || '127.0.0.1')
    const ua = req.headers.get('user-agent') || 'Unknown'

    // Update active session snapshot
    await prisma.session.upsert({
      where: { token: userId },
      update: {
        currentPage: page || '/dashboard',
        currentLesson: lessonTitle || null,
        lastSeen: new Date(),
        isActive: true,
        ip: clientIp,
        userAgent: ua,
      },
      create: {
        userId,
        token: userId,
        currentPage: page || '/dashboard',
        currentLesson: lessonTitle || null,
        lastSeen: new Date(),
        isActive: true,
        ip: clientIp,
        userAgent: ua,
        expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000),
      },
    })

    // Log Activity record if action specified
    if (action) {
      await prisma.activityLog.create({
        data: {
          userId,
          action,
          page: page || null,
          lessonId: lessonId || null,
          metadata: metadata ? JSON.stringify(metadata) : null,
          ip: clientIp,
          userAgent: ua,
        },
      })
    }

    return NextResponse.json({ success: true, timestamp: new Date() })
  } catch (error: any) {
    console.error('Tracking error:', error)
    return NextResponse.json({ error: 'Error al registrar actividad' }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prismaDash } from '@/lib/prisma-dashboard'
import { parseUA } from '@/lib/useragent'
import { lookupGeo } from '@/lib/geo'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

export async function GET() {
  try {
    const s: any = await getServerSession(authOptions)
    if (!s?.user || s.user.role !== 'ADMIN') return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const now = Date.now()
    const onlineSince = new Date(now - 3 * 60 * 1000)
    const daySince = new Date(now - 24 * 3600 * 1000)

    const sessions = await prismaDash.session.findMany({
      where: { lastSeen: { gte: daySince }, token: { startsWith: 'web-' } },
      orderBy: { lastSeen: 'desc' },
      take: 300,
      include: { user: { select: { id: true, name: true, email: true, role: true, avatarUrl: true } } },
    })

    const ips = Array.from(new Set(sessions.map((x) => x.ip).filter(Boolean) as string[])).slice(0, 60)
    const geoMap = new Map<string, Awaited<ReturnType<typeof lookupGeo>>>()
    await Promise.all(ips.map(async (ip) => geoMap.set(ip, await lookupGeo(ip))))

    const lessonIds = Array.from(new Set(sessions.map((x) => x.currentLesson).filter(Boolean) as string[]))
    const events = await prismaDash.activityLog.findMany({
      orderBy: { createdAt: 'desc' },
      take: 50,
      include: { user: { select: { name: true, email: true, role: true } }, lesson: { select: { title: true } } },
    })
    const lessons = lessonIds.length
      ? await prismaDash.lesson.findMany({ where: { id: { in: lessonIds } }, select: { id: true, title: true } })
      : []
    const lessonTitle = new Map(lessons.map((l) => [l.id, l.title]))

    const rows = sessions.map((x) => {
      const ua = parseUA(x.userAgent)
      const g = x.ip ? geoMap.get(x.ip) ?? null : null
      return {
        id: x.id,
        online: x.lastSeen >= onlineSince,
        lastSeen: x.lastSeen,
        page: x.currentPage,
        lesson: x.currentLesson ? lessonTitle.get(x.currentLesson) ?? null : null,
        ip: x.ip,
        ...ua,
        country: g?.country ?? null,
        countryCode: g?.countryCode ?? null,
        city: g?.city ?? null,
        lat: g?.lat ?? null,
        lon: g?.lon ?? null,
        user: x.user,
      }
    })

    const count = (arr: (string | null)[]) => {
      const m = new Map<string, number>()
      arr.forEach((k) => { const key = k || 'Desconocido'; m.set(key, (m.get(key) || 0) + 1) })
      return Array.from(m, ([name, value]) => ({ name, value })).sort((a, b) => b.value - a.value)
    }

    const security = await prismaDash.securityLog.findMany({ orderBy: { createdAt: 'desc' }, take: 40 })
    const uniqueUsers24 = new Set(sessions.map((x) => x.userId)).size
    const onlineUsers = new Set(rows.filter((r) => r.online).map((r) => r.user.id)).size

    return NextResponse.json({
      stats: { onlineUsers, uniqueUsers24, events: events.length, alerts: security.length },
      sessions: rows,
      browsers: count(rows.map((r) => r.browser)),
      systems: count(rows.map((r) => r.os)),
      devices: count(rows.map((r) => r.device)),
      countries: count(rows.map((r) => r.country)),
      events: events.map((e) => ({
        id: e.id, action: e.action, page: e.page, createdAt: e.createdAt, ip: e.ip,
        user: e.user, lesson: e.lesson?.title ?? null,
      })),
      security: security.map((x) => ({ id: x.id, action: x.action, email: x.email, details: x.details, ip: x.ipAddress, createdAt: x.createdAt })),
    })
  } catch (err: any) {
    console.error('live-data:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

async function safeJson(res: Response): Promise<any> {
  const text = await res.text()
  if (!text) return {}

// Traduce errores técnicos de la subida a un motivo y una solución
function explainUploadError(raw: string, file: File): { title: string; detail: string } {
  const m = raw.toLowerCase()
  const mb = (file.size / 1024 / 1024).toFixed(1)
  if (m.includes('exceeded') || m.includes('maximum allowed size') || m.includes('too large') || m.includes('413')) {
    return {
      title: 'El video es demasiado pesado',
      detail: `Tu archivo pesa ${mb} MB y Supabase solo admite hasta ${MAX_MB} MB en este plan. Comprímelo (por ejemplo con HandBrake), o súbelo a YouTube/Vimeo y pega el enlace.`,
    }
  }
  if (m.includes('bucket') && m.includes('not found')) {
    return { title: 'No existe el bucket "videos"', detail: 'En Supabase → Storage crea un bucket llamado exactamente "videos" y márcalo como público.' }
  }
  if (m.includes('mime') || m.includes('not supported') || m.includes('invalid file')) {
    return { title: 'Formato de video no admitido', detail: 'Usa MP4, WebM, MKV o MOV. Lo más compatible es MP4 (H.264).' }
  }
  if (m.includes('failed to fetch') || m.includes('network') || m.includes('load failed')) {
    return { title: 'Se cortó la conexión durante la subida', detail: 'Revisa tu internet y vuelve a intentarlo. Con videos grandes, una conexión inestable suele cortar la subida.' }
  }
  if (m.includes('invalid') && (m.includes('jwt') || m.includes('key') || m.includes('signature') || m.includes('token'))) {
    return { title: 'Clave de Supabase incorrecta o vencida', detail: 'Revisa NEXT_PUBLIC_SUPABASE_ANON_KEY y SUPABASE_SERVICE_ROLE_KEY en Netlify y haz un redeploy.' }
  }
  if (m.includes('row-level security') || m.includes('policy') || m.includes('unauthorized')) {
    return { title: 'Supabase rechazó la subida por permisos', detail: 'La URL de subida venció o el bucket tiene políticas restrictivas. Vuelve a intentarlo; si persiste, revisa las políticas del bucket "videos".' }
  }
  return { title: 'No se pudo subir el video', detail: raw }
}

interface Lesson {
  id: string; title: string; description: string | null
  duration: number | null; order: number; allowedRole: string

interface Mod { id: string; title: string; description: string | null; lessons: Lesson[] }

const STORE_KEY = 'invictus:lastLesson'

function fmtDuration(s: number | null) {
  if (!s) return null
  const m = Math.round(s / 60)
  return m >= 60 ? `${Math.floor(m / 60)} h ${m % 60} min` : `${m} min`
}

function CourseContent({
  modules, progress, currentId, openMods, onToggle, onOpen,
}: {
  modules: Mod[]; progress: Record<string, boolean>; currentId: string | null
  openMods: Set<string>; onToggle: (id: string) => void; onOpen: (id: string) => void
}) {
  return (
    <div className="space-y-3">
      {modules.map((mod, i) => {
        const done = mod.lessons.filter((l) => progress[l.id]).length
        const pct = Math.round((done / mod.lessons.length) * 100)
        const open = openMods.has(mod.id)
        return (
          <div key={mod.id} className="overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-sm">
            <button type="button" onClick={() => onToggle(mod.id)} className="flex w-full items-center gap-3 px-4 py-3.5 text-left cursor-pointer">
              <span className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-xl text-xs font-black ${pct === 100 ? 'bg-emerald-100 text-emerald-600' : 'bg-sky-100 text-sky-600'}`}>
                {pct === 100 ? <Check size={17} /> : i + 1}
              </span>
              <div className="min-w-0 flex-1">
                <p className="truncate text-sm font-bold text-slate-900">{mod.title}</p>
                <p className="text-[11px] text-slate-500">{done} de {mod.lessons.length} completadas</p>
                <div className="mt-1.5 h-1.5 overflow-hidden rounded-full bg-slate-100">
                  <div className={`h-full rounded-full transition-all duration-500 ${pct === 100 ? 'bg-emerald-500' : 'bg-sky-500'}`} style={{ width: `${pct}%` }} />
                </div>
              </div>
              <ChevronDown size={18} className={`shrink-0 text-slate-400 transition-transform ${open ? 'rotate-180' : ''}`} />
            </button>

            {open && (
              <div className="space-y-1 border-t border-slate-100 bg-slate-50/60 p-2">
                {mod.lessons.map((l) => {
                  const completed = !!progress[l.id]
                  const current = l.id === currentId
                  const d = fmtDuration(l.duration)
                  return (
                    <button key={l.id} type="button" onClick={() => onOpen(l.id)}
                      className={`flex w-full items-center gap-3 rounded-xl px-3 py-2.5 text-left transition-all cursor-pointer ${current ? 'bg-sky-50 ring-1 ring-sky-300' : 'hover:bg-white'}`}>
                      {completed ? <CheckCircle2 size={19} className="shrink-0 text-emerald-500" />
                        : current ? <PlayCircle size={19} className="shrink-0 text-sky-500" />
                        : <Circle size={19} className="shrink-0 text-slate-300" />}
                      <div className="min-w-0 flex-1">

                    </button>
                  )

              </div>
            )}
          </div>
        )
      })}
    </div>
  )
}

export default function ClasesPage() {
  const [modules, setModules] = useState<Mod[]>([])
  const [progress, setProgress] = useState<Record<string, boolean>>({})
  const [role, setRole] = useState('CLUB')
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)
  const [view, setView] = useState<'list' | 'lesson'>('list')
  const [currentId, setCurrentId] = useState<string | null>(null)
  const [openMods, setOpenMods] = useState<Set<string>>(new Set())
  const [firstLoad, setFirstLoad] = useState(true)

  const load = useCallback(async () => {
    try {
      const res = await fetch('/api/student/courses', { cache: 'no-store' })
      if (res.status === 401) { window.location.href = '/login'; return }
      const data = await res.json()
      if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`)
      setModules(data.modules)
      setRole(data.role)
      const p: Record<string, boolean> = {}
      Object.entries<any>(data.progress).forEach(([k, v]) => { p[k] = !!v.completed })
      setProgress(p)
      setError(null)
    } catch (e: any) {
      setError(e?.message || 'No se pudo cargar el contenido')
    } finally {
      setLoading(false)
    }
  }, [])

  // Sincronización: cada 20 s y al volver a la pestaña
  useEffect(() => {
    load()
    const t = setInterval(load, 20000)
    const onVis = () => { if (document.visibilityState === 'visible') load() }
    document.addEventListener('visibilitychange', onVis)
    return () => { clearInterval(t); document.removeEventListener('visibilitychange', onVis) }
  }, [load])

  const flat = useMemo(() => modules.flatMap((m) => m.lessons), [modules])

  // Primera carga: abre el módulo de la última lección, o el primero
  useEffect(() => {
    if (!firstLoad || modules.length === 0) return
    let last: string | null = null
    try { last = localStorage.getItem(STORE_KEY) } catch {}
    const target = flat.find((l) => l.id === last) ?? flat.find((l) => !progress[l.id]) ?? flat[0]
    if (target) {
      setCurrentId(target.id)
      const mod = modules.find((m) => m.lessons.some((l) => l.id === target.id))
      setOpenMods(new Set(mod ? [mod.id] : [modules[0].id]))
    }
    setFirstLoad(false)
  }, [firstLoad, modules, flat, progress])

  // Si la lección actual fue eliminada por el admin
  useEffect(() => {
    if (currentId && flat.length && !flat.some((l) => l.id === currentId)) {
      setCurrentId(flat[0].id)
    }
    if (flat.length === 0 && view === 'lesson') setView('list')
  }, [flat, currentId, view])

  const current = flat.find((l) => l.id === currentId) ?? null
  const idx = current ? flat.findIndex((l) => l.id === current.id) : -1
  const currentModule = current ? modules.find((m) => m.lessons.some((l) => l.id === current.id)) : null
  const total = flat.length
  const done = flat.filter((l) => progress[l.id]).length
  const pct = total ? Math.round((done / total) * 100) : 0
  const m = getMembership(role)
  const MIcon = m.key === 'MENTORIA' ? Crown : m.key === 'ADMIN' ? ShieldCheck : Star

  const toggleModule = (id: string) => setOpenMods((prev) => { const n = new Set(prev); n.has(id) ? n.delete(id) : n.add(id); return n })

  const openLesson = (id: string) => {
    setCurrentId(id)
    setView('lesson')
    try { localStorage.setItem(STORE_KEY, id) } catch {}    try { sessionStorage.setItem('invictus:viewing', id) } catch {}
    const mod = modules.find((x) => x.lessons.some((l) => l.id === id))
    if (mod) setOpenMods((prev) => new Set(prev).add(mod.id))
    window.scrollTo({ top: 0, behavior: 'smooth' })
  }

  const toggleComplete = async () => {
    if (!current) return
    const next = !progress[current.id]
    setProgress((p) => ({ ...p, [current.id]: next }))
    try {
      const res = await fetch('/api/student/progress', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ lessonId: current.id, completed: next }),
      })
      if (!res.ok) throw new Error()
    } catch {
      setProgress((p) => ({ ...p, [current.id]: !next }))
      setError('No se pudo guardar tu progreso. Revisa tu conexión e inténtalo de nuevo.')
    }
  }

  if (loading) {
    return <div className="flex min-h-[60vh] items-center justify-center text-xs font-bold text-slate-400">Cargando tus clases...</div>
  }

  return (
    <div className="mx-auto max-w-6xl space-y-6 p-5 md:p-8">
      {error && (
        <div className="flex items-center gap-2 rounded-2xl border border-rose-200 bg-rose-50 p-3.5 text-xs font-bold text-rose-700">
          <AlertCircle size={16} className="shrink-0" /> {error}
        </div>
      )}

      {total === 0 ? (
        <div className="rounded-3xl border border-dashed border-slate-300 bg-white p-12 text-center">
          <BookOpen size={32} className="mx-auto mb-3 text-slate-300" />
          <p className="text-sm font-bold text-slate-700">Aún no hay clases disponibles para tu membresía</p>
          <p className="mt-1 text-xs text-slate-400">Cuando se publiquen, aparecerán aquí automáticamente.</p>
        </div>
      ) : view === 'list' ? (
        <>
          {/* Banner de membresía */}
          <div className={`relative overflow-hidden rounded-3xl bg-gradient-to-br ${m.gradient} p-7 text-white shadow-xl ${m.glow}`}>
            <MIcon size={150} className="absolute -right-6 -top-6 opacity-15" />
            <div className="relative flex flex-col gap-5 md:flex-row md:items-center md:justify-between">
              <div>
                <span className="inline-flex items-center gap-1.5 rounded-full bg-white/20 px-3 py-1 text-[11px] font-black uppercase tracking-widest backdrop-blur">
                  <MIcon size={13} /> {m.title}
                </span>
                <h1 className="mt-3 text-2xl font-black tracking-tight sm:text-3xl">Tus lecciones</h1>
                <p className="mt-1 max-w-md text-sm text-white/85">{m.tagline}</p>
              </div>
              <button type="button" onClick={() => openLesson(currentId ?? flat[0].id)}
                className="inline-flex items-center justify-center gap-2 rounded-full bg-white px-6 py-3.5 text-sm font-black text-slate-900 shadow-lg transition-transform hover:scale-105 cursor-pointer">
                <PlayCircle size={18} /> {done === 0 ? 'Empezar' : 'Continuar donde lo dejaste'}
              </button>
            </div>
          </div>

          {/* Resumen de progreso */}
          <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
            {[
              { label: 'Progreso', value: `${pct}%`, icon: Sparkles, color: 'text-sky-600 bg-sky-50' },
              { label: 'Completadas', value: done, icon: CheckCircle2, color: 'text-emerald-600 bg-emerald-50' },
              { label: 'Por ver', value: total - done, icon: Clock, color: 'text-amber-600 bg-amber-50' },
              { label: 'Total', value: total, icon: BookOpen, color: 'text-slate-600 bg-slate-100' },
            ].map((s) => (
              <div key={s.label} className="flex items-center gap-3 rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
                <span className={`flex h-10 w-10 items-center justify-center rounded-xl ${s.color}`}><s.icon size={19} /></span>
                <div>
                  <p className="text-xl font-black leading-none text-slate-900">{s.value}</p>
                  <p className="mt-1 text-[11px] font-bold text-slate-500">{s.label}</p>
                </div>
              </div>
            ))}
          </div>
          <div className="h-2.5 overflow-hidden rounded-full bg-slate-200">
            <div className="h-full rounded-full bg-gradient-to-r from-sky-500 to-cyan-400 transition-all duration-700" style={{ width: `${pct}%` }} />
          </div>

          <div>
            <h2 className="mb-3 text-sm font-black text-slate-900">Contenido del curso</h2>
            <CourseContent modules={modules} progress={progress} currentId={currentId} openMods={openMods} onToggle={toggleModule} onOpen={openLesson} />
          </div>
        </>
      ) : (
        current && (
          <>
            <button type="button" onClick={() => { setView('list'); try { sessionStorage.removeItem('invictus:viewing') } catch {} }} className="flex items-center gap-1.5 text-xs font-bold text-slate-500 hover:text-sky-600 cursor-pointer">
              <ArrowLeft size={15} /> Volver a las lecciones
            </button>

            <div className="grid grid-cols-1 gap-6 xl:grid-cols-3">
              <div className="space-y-4 xl:col-span-2">
                <div>
                  <p className="text-[11px] font-black uppercase tracking-widest text-sky-500">{currentModule?.title}</p>
                  <h1 className="mt-1 text-xl font-black tracking-tight text-slate-900">{current.title}</h1>
                </div>

                <SecurePlayer key={current.id} lessonId={current.id} title={current.title} />

                <div className="flex flex-wrap items-center justify-between gap-3">
                  <button type="button" onClick={toggleComplete}
                    className={`inline-flex items-center gap-2 rounded-full px-5 py-2.5 text-xs font-black transition-all cursor-pointer ${
                      progress[current.id] ? 'bg-emerald-100 text-emerald-700 hover:bg-emerald-200' : 'bg-sky-600 text-white shadow-md shadow-sky-300/50 hover:bg-sky-700'
                    }`}>
                    <CheckCircle2 size={16} /> {progress[current.id] ? 'Completada · clic para desmarcar' : 'Marcar como completada'}
                  </button>
                  <div className="flex gap-2">
                    <button type="button" disabled={idx <= 0} onClick={() => openLesson(flat[idx - 1].id)}
                      className="inline-flex items-center gap-1 rounded-full border border-slate-200 bg-white px-4 py-2.5 text-xs font-bold text-slate-700 hover:border-sky-300 disabled:opacity-40 cursor-pointer">
                      <ChevronLeft size={15} /> Anterior
                    </button>
                    <button type="button" disabled={idx >= flat.length - 1} onClick={() => openLesson(flat[idx + 1].id)}
                      className="inline-flex items-center gap-1 rounded-full border border-slate-200 bg-white px-4 py-2.5 text-xs font-bold text-slate-700 hover:border-sky-300 disabled:opacity-40 cursor-pointer">
                      Siguiente <ChevronRight size={15} />
                    </button>
                  </div>
                </div>

                <div className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
                  <h2 className="text-sm font-black text-slate-900">Sobre esta clase</h2>
                  <p className="mt-2 text-sm leading-relaxed text-slate-600">{current.description || 'Esta clase no tiene descripción.'}</p>
                </div>
              </div>

              <div className="space-y-3 xl:sticky xl:top-6 xl:self-start">
                <div className="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
                  <div className="flex items-center justify-between text-xs font-black text-slate-700">
                    <span>Tu progreso</span><span className="text-sky-600">{pct}%</span>
                  </div>
                  <div className="mt-2 h-2 overflow-hidden rounded-full bg-slate-100">
                    <div className="h-full rounded-full bg-gradient-to-r from-sky-500 to-cyan-400 transition-all duration-700" style={{ width: `${pct}%` }} />
                  </div>
                  <p className="mt-1.5 text-[11px] text-slate-500">{done} de {total} clases completadas</p>
                </div>
                <CourseContent modules={modules} progress={progress} currentId={currentId} openMods={openMods} onToggle={toggleModule} onOpen={openLesson} />
              </div>
            </div>
          </>
        )
      )}
    </div>
  )
}
