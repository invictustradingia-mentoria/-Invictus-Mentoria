'use client'

import { useCallback, useEffect, useMemo, useState } from 'react'
import { CheckCircle2, Circle, PlayCircle, ChevronDown, ChevronLeft, ChevronRight, ArrowLeft, Check, Crown, Star, ShieldCheck, Sparkles, Clock, BookOpen, AlertCircle, Pencil } from 'lucide-react'
import SecurePlayer from '@/components/dashboard/SecurePlayer'
import { getMembership } from '@/lib/membership'

interface Lesson {
  id: string; title: string; description: string | null
  duration: number | null; order: number; allowedRole: string
}
interface Mod { id: string; title: string; description: string | null; lessons: Lesson[] }

const STORE_KEY = 'invictus:lastLesson'

function fmtDuration(s: number | null) {
  if (!s) return null
  const m = Math.round(s / 60)
  return m >= 60 ? `${Math.floor(m / 60)} h ${m % 60} min` : `${m} min`
}

function CourseContent({
  modules, progress, currentId, openMods, onToggle, onOpen,
}: {
  modules: Mod[]; progress: Record<string, boolean>; currentId: string | null
  openMods: Set<string>; onToggle: (id: string) => void; onOpen: (id: string) => void
}) {
  return (
    <div className="space-y-3">
      {modules.map((mod, i) => {
        const done = mod.lessons.filter((l) => progress[l.id]).length
        const pct = Math.round((done / mod.lessons.length) * 100)
        const open = openMods.has(mod.id)
        return (
          <div key={mod.id} className="overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-sm">
            <button type="button" onClick={() => onToggle(mod.id)} className="flex w-full items-center gap-3 px-4 py-3.5 text-left cursor-pointer">
              <span className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-xl text-xs font-black ${pct === 100 ? 'bg-emerald-100 text-emerald-600' : 'bg-sky-100 text-sky-600'}`}>
                {pct === 100 ? <Check size={17} /> : i + 1}
              </span>
              <div className="min-w-0 flex-1">
                <p className="truncate text-sm font-bold text-slate-900">{mod.title}</p>
                <p className="text-[11px] text-slate-500">{done} de {mod.lessons.length} completadas</p>
                <div className="mt-1.5 h-1.5 overflow-hidden rounded-full bg-slate-100">
                  <div className={`h-full rounded-full transition-all duration-500 ${pct === 100 ? 'bg-emerald-500' : 'bg-sky-500'}`} style={{ width: `${pct}%` }} />
                </div>
              </div>
              <ChevronDown size={18} className={`shrink-0 text-slate-400 transition-transform ${open ? 'rotate-180' : ''}`} />
            </button>

            {open && (
              <div className="space-y-1 border-t border-slate-100 bg-slate-50/60 p-2">
                {mod.lessons.map((l) => {
                  const completed = !!progress[l.id]
                  const current = l.id === currentId
                  const d = fmtDuration(l.duration)
                  return (
                    <button key={l.id} type="button" onClick={() => onOpen(l.id)}
                      className={`flex w-full items-center gap-3 rounded-xl px-3 py-2.5 text-left transition-all cursor-pointer ${current ? 'bg-sky-50 ring-1 ring-sky-300' : 'hover:bg-white'}`}>
                      {completed ? <CheckCircle2 size={19} className="shrink-0 text-emerald-500" />
                        : current ? <PlayCircle size={19} className="shrink-0 text-sky-500" />
                        : <Circle size={19} className="shrink-0 text-slate-300" />}
                      <div className="min-w-0 flex-1">
                        <p className={`truncate text-xs font-bold ${completed ? 'text-slate-500' : 'text-slate-800'}`}>{l.title}</p>
                        <p className="text-[10px] text-slate-400">
                          {completed ? 'Completada' : current ? 'Viendo ahora' : 'Pendiente'}{d ? ` · ${d}` : ''}
                        </p>
                      </div>
                      {l.allowedRole === 'MENTORIA' && <span className="shrink-0 rounded-md bg-amber-100 px-1.5 py-0.5 text-[9px] font-black uppercase text-amber-700">Mentoría</span>}
                    </button>
                  )
                })}
              </div>
            )}
          </div>
        )
      })}
    </div>
  )
}

export default function ClasesPage() {
  const [modules, setModules] = useState<Mod[]>([])
  const [progress, setProgress] = useState<Record<string, boolean>>({})
  const [role, setRole] = useState('CLUB')
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)
  const [view, setView] = useState<'list' | 'lesson'>('list')
  const [currentId, setCurrentId] = useState<string | null>(null)
  const [openMods, setOpenMods] = useState<Set<string>>(new Set())
  const [firstLoad, setFirstLoad] = useState(true)

  const load = useCallback(async () => {
    try {
      const res = await fetch('/api/student/courses', { cache: 'no-store' })
      if (res.status === 401) { window.location.href = '/login'; return }
      const data = await res.json()
      if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`)
      setModules(data.modules)
      setRole(data.role)
      const p: Record<string, boolean> = {}
      Object.entries<any>(data.progress).forEach(([k, v]) => { p[k] = !!v.completed })
      setProgress(p)
      setError(null)
    } catch (e: any) {
      setError(e?.message || 'No se pudo cargar el contenido')
    } finally {
      setLoading(false)
    }
  }, [])

  // Sincronización: cada 20 s y al volver a la pestaña
  useEffect(() => {
    load()
    const t = setInterval(load, 20000)
    const onVis = () => { if (document.visibilityState === 'visible') load() }
    document.addEventListener('visibilitychange', onVis)
    return () => { clearInterval(t); document.removeEventListener('visibilitychange', onVis) }
  }, [load])

  const flat = useMemo(() => modules.flatMap((m) => m.lessons), [modules])

  // Primera carga: abre el módulo de la última lección, o el primero
  useEffect(() => {
    if (!firstLoad || modules.length === 0) return
    let last: string | null = null
    try { last = localStorage.getItem(STORE_KEY) } catch {}
    const target = flat.find((l) => l.id === last) ?? flat.find((l) => !progress[l.id]) ?? flat[0]
    if (target) {
      setCurrentId(target.id)
      const mod = modules.find((m) => m.lessons.some((l) => l.id === target.id))
      setOpenMods(new Set(mod ? [mod.id] : [modules[0].id]))
    }
    setFirstLoad(false)
  }, [firstLoad, modules, flat, progress])

  // Si la lección actual fue eliminada por el admin
  useEffect(() => {
    if (currentId && flat.length && !flat.some((l) => l.id === currentId)) {
      setCurrentId(flat[0].id)
    }
    if (flat.length === 0 && view === 'lesson') setView('list')
  }, [flat, currentId, view])

  const current = flat.find((l) => l.id === currentId) ?? null
  const idx = current ? flat.findIndex((l) => l.id === current.id) : -1
  const currentModule = current ? modules.find((m) => m.lessons.some((l) => l.id === current.id)) : null
  const total = flat.length
  const done = flat.filter((l) => progress[l.id]).length
  const pct = total ? Math.round((done / total) * 100) : 0
  const m = getMembership(role)
  const MIcon = m.key === 'MENTORIA' ? Crown : m.key === 'ADMIN' ? ShieldCheck : m.key === 'EDITOR' ? Pencil : Star

  const toggleModule = (id: string) => setOpenMods((prev) => { const n = new Set(prev); n.has(id) ? n.delete(id) : n.add(id); return n })

  const openLesson = (id: string) => {
    setCurrentId(id)
    setView('lesson')
    try { localStorage.setItem(STORE_KEY, id) } catch {}    try { sessionStorage.setItem('invictus:viewing', id) } catch {}
    const mod = modules.find((x) => x.lessons.some((l) => l.id === id))
    if (mod) setOpenMods((prev) => new Set(prev).add(mod.id))
    window.scrollTo({ top: 0, behavior: 'smooth' })
  }

  const toggleComplete = async () => {
    if (!current) return
    const next = !progress[current.id]
    setProgress((p) => ({ ...p, [current.id]: next }))
    try {
      const res = await fetch('/api/student/progress', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ lessonId: current.id, completed: next }),
      })
      if (!res.ok) throw new Error()
    } catch {
      setProgress((p) => ({ ...p, [current.id]: !next }))
      setError('No se pudo guardar tu progreso. Revisa tu conexión e inténtalo de nuevo.')
    }
  }

  if (loading) {
    return <div className="flex min-h-[60vh] items-center justify-center text-xs font-bold text-slate-400">Cargando tus clases...</div>
  }

  return (
    <div className="mx-auto max-w-6xl space-y-6 p-5 md:p-8">
      {error && (
        <div className="flex items-center gap-2 rounded-2xl border border-rose-200 bg-rose-50 p-3.5 text-xs font-bold text-rose-700">
          <AlertCircle size={16} className="shrink-0" /> {error}
        </div>
      )}

      {total === 0 ? (
        <div className="rounded-3xl border border-dashed border-slate-300 bg-white p-12 text-center">
          <BookOpen size={32} className="mx-auto mb-3 text-slate-300" />
          <p className="text-sm font-bold text-slate-700">Aún no hay clases disponibles para tu membresía</p>
          <p className="mt-1 text-xs text-slate-400">Cuando se publiquen, aparecerán aquí automáticamente.</p>
        </div>
      ) : view === 'list' ? (
        <>
          {/* Banner de membresía */}
          <div className={`relative overflow-hidden rounded-3xl bg-gradient-to-br ${m.gradient} p-7 text-white shadow-xl ${m.glow}`}>
            <MIcon size={150} className="absolute -right-6 -top-6 opacity-15" />
            <div className="relative flex flex-col gap-5 md:flex-row md:items-center md:justify-between">
              <div>
                <span className="inline-flex items-center gap-1.5 rounded-full bg-white/20 px-3 py-1 text-[11px] font-black uppercase tracking-widest backdrop-blur">
                  <MIcon size={13} /> {m.title}
                </span>
                <h1 className="mt-3 text-2xl font-black tracking-tight sm:text-3xl">Tus lecciones</h1>
                <p className="mt-1 max-w-md text-sm text-white/85">{m.tagline}</p>
              </div>
              <button type="button" onClick={() => openLesson(currentId ?? flat[0].id)}
                className="inline-flex items-center justify-center gap-2 rounded-full bg-white px-6 py-3.5 text-sm font-black text-slate-900 shadow-lg transition-transform hover:scale-105 cursor-pointer">
                <PlayCircle size={18} /> {done === 0 ? 'Empezar' : 'Continuar donde lo dejaste'}
              </button>
            </div>
          </div>

          {/* Resumen de progreso */}
          <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
            {[
              { label: 'Progreso', value: `${pct}%`, icon: Sparkles, color: 'text-sky-600 bg-sky-50' },
              { label: 'Completadas', value: done, icon: CheckCircle2, color: 'text-emerald-600 bg-emerald-50' },
              { label: 'Por ver', value: total - done, icon: Clock, color: 'text-amber-600 bg-amber-50' },
              { label: 'Total', value: total, icon: BookOpen, color: 'text-slate-600 bg-slate-100' },
            ].map((s) => (
              <div key={s.label} className="flex items-center gap-3 rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
                <span className={`flex h-10 w-10 items-center justify-center rounded-xl ${s.color}`}><s.icon size={19} /></span>
                <div>
                  <p className="text-xl font-black leading-none text-slate-900">{s.value}</p>
                  <p className="mt-1 text-[11px] font-bold text-slate-500">{s.label}</p>
                </div>
              </div>
            ))}
          </div>
          <div className="h-2.5 overflow-hidden rounded-full bg-slate-200">
            <div className="h-full rounded-full bg-gradient-to-r from-sky-500 to-cyan-400 transition-all duration-700" style={{ width: `${pct}%` }} />
          </div>

          <div>
            <h2 className="mb-3 text-sm font-black text-slate-900">Contenido del curso</h2>
            <CourseContent modules={modules} progress={progress} currentId={currentId} openMods={openMods} onToggle={toggleModule} onOpen={openLesson} />
          </div>
        </>
      ) : (
        current && (
          <>
            <button type="button" onClick={() => { setView('list'); try { sessionStorage.removeItem('invictus:viewing') } catch {} }} className="flex items-center gap-1.5 text-xs font-bold text-slate-500 hover:text-sky-600 cursor-pointer">
              <ArrowLeft size={15} /> Volver a las lecciones
            </button>

            <div className="grid grid-cols-1 gap-6 xl:grid-cols-3">
              <div className="space-y-4 xl:col-span-2">
                <div>
                  <p className="text-[11px] font-black uppercase tracking-widest text-sky-500">{currentModule?.title}</p>
                  <h1 className="mt-1 text-xl font-black tracking-tight text-slate-900">{current.title}</h1>
                </div>

                <SecurePlayer key={current.id} lessonId={current.id} title={current.title} />

                <div className="flex flex-wrap items-center justify-between gap-3">
                  <button type="button" onClick={toggleComplete}
                    className={`inline-flex items-center gap-2 rounded-full px-5 py-2.5 text-xs font-black transition-all cursor-pointer ${
                      progress[current.id] ? 'bg-emerald-100 text-emerald-700 hover:bg-emerald-200' : 'bg-sky-600 text-white shadow-md shadow-sky-300/50 hover:bg-sky-700'
                    }`}>
                    <CheckCircle2 size={16} /> {progress[current.id] ? 'Completada · clic para desmarcar' : 'Marcar como completada'}
                  </button>
                  <div className="flex gap-2">
                    <button type="button" disabled={idx <= 0} onClick={() => openLesson(flat[idx - 1].id)}
                      className="inline-flex items-center gap-1 rounded-full border border-slate-200 bg-white px-4 py-2.5 text-xs font-bold text-slate-700 hover:border-sky-300 disabled:opacity-40 cursor-pointer">
                      <ChevronLeft size={15} /> Anterior
                    </button>
                    <button type="button" disabled={idx >= flat.length - 1} onClick={() => openLesson(flat[idx + 1].id)}
                      className="inline-flex items-center gap-1 rounded-full border border-slate-200 bg-white px-4 py-2.5 text-xs font-bold text-slate-700 hover:border-sky-300 disabled:opacity-40 cursor-pointer">
                      Siguiente <ChevronRight size={15} />
                    </button>
                  </div>
                </div>

                <div className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
                  <h2 className="text-sm font-black text-slate-900">Sobre esta clase</h2>
                  <p className="mt-2 text-sm leading-relaxed text-slate-600">{current.description || 'Esta clase no tiene descripción.'}</p>

              <div className="space-y-3 xl:sticky xl:top-6 xl:self-start">
                <div className="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
                  <div className="flex items-center justify-between text-xs font-black text-slate-700">

export async function POST(req: Request) {

    const name = typeof body.name === 'string' ? body.name.trim() : ''
    const email = typeof body.email === 'string' ? body.email.trim().toLowerCase() : ''
    const password = typeof body.password === 'string' ? body.password : ''

    const exists = await prismaDash.user.findUnique({ where: { email } })

      return NextResponse.json({ error: 'Ya existe una cuenta con ese correo' }, { status: 409 })
    }

    const ip = req.headers.get('x-nf-client-connection-ip') || req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() || null

        isBanned: false,

        userAgent,
      },
    })

      success: true,

import { NextResponse } from 'next/server'

export const runtime = 'nodejs'
export const dynamic = 'force-dynamic'

const AVATAR_RE = /^data:image\/(jpeg|png|webp);base64,[A-Za-z0-9+/=]+$/
const MAX_AVATAR_CHARS = 300_000

export async function GET() {
  try {
    const session: any = await getServerSession(authOptions)
    const id = session?.user?.id
    if (!id) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const user = await prismaDash.user.findUnique({
      where: { id },
      select: { name: true, email: true, phone: true, role: true, avatarUrl: true, createdAt: true },
    })
    if (!user) return NextResponse.json({ error: 'Usuario no encontrado' }, { status: 404 })
    return NextResponse.json({ user })
  } catch (err: any) {
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

export async function PATCH(req: Request) {
  try {
    const session: any = await getServerSession(authOptions)
    const id = session?.user?.id
    if (!id) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

    const body = await req.json()
    const data: Record<string, any> = {}

    if (typeof body.name === 'string') {
      const n = body.name.trim()
      if (!n) return NextResponse.json({ error: 'El nombre no puede estar vacío' }, { status: 400 })
      data.name = n.slice(0, 80)
    }
    if (typeof body.phone === 'string') data.phone = body.phone.trim().slice(0, 30) || null

    if (body.avatarUrl === null) {
      data.avatarUrl = null
    } else if (typeof body.avatarUrl === 'string') {
      if (!AVATAR_RE.test(body.avatarUrl) || body.avatarUrl.length > MAX_AVATAR_CHARS) {
        return NextResponse.json({ error: 'La imagen no es válida o es demasiado pesada' }, { status: 400 })
      }
      data.avatarUrl = body.avatarUrl
    }

    if (typeof body.newPassword === 'string' && body.newPassword) {
      if (body.newPassword.length < 8) {
        return NextResponse.json({ error: 'La nueva contraseña debe tener al menos 8 caracteres' }, { status: 400 })
      }
      const current = await prismaDash.user.findUnique({ where: { id }, select: { password: true } })
      const ok = current && typeof body.currentPassword === 'string' && (await bcrypt.compare(body.currentPassword, current.password))
      if (!ok) return NextResponse.json({ error: 'La contraseña actual no es correcta' }, { status: 400 })
      data.password = await bcrypt.hash(body.newPassword, 10)
    }

    if (Object.keys(data).length === 0) {
      return NextResponse.json({ error: 'No hay cambios para guardar' }, { status: 400 })
    }

    const user = await prismaDash.user.update({
      where: { id },
      data,
      select: { name: true, email: true, phone: true, role: true, avatarUrl: true },
    })
    return NextResponse.json({ success: true, user })
  } catch (err: any) {
    console.error('PATCH student/profile:', err)
    return NextResponse.json({ error: err?.message || 'Error interno' }, { status: 500 })
  }
}

    if (!sid.current) return
    let lessonId: string | null = null
    try { lessonId = sessionStorage.getItem('invictus:viewing') } catch {}
    fetch('/api/student/ping', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ sid: sid.current, page: window.location.pathname, lessonId, event }),
      keepalive: true,
    }).catch(() => {})
  }

  useEffect(() => {
    try {
      let s = sessionStorage.getItem('invictus:sid')
      if (!s) { s = crypto.randomUUID(); sessionStorage.setItem('invictus:sid', s) }
      sid.current = s
    } catch { sid.current = Math.random().toString(36).slice(2) }

    const beat = setInterval(() => { if (document.visibilityState === 'visible') send() }, 25000)
    const watch = setInterval(() => {
      let cur: string | null = null
      try { cur = sessionStorage.getItem('invictus:viewing') } catch {}
      if (cur !== lastLesson.current) { lastLesson.current = cur; if (cur) send('LESSON_VIEW'); else send() }
    }, 3000)
    return () => { clearInterval(beat); clearInterval(watch) }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  useEffect(() => {
    if (sid.current) send('PAGE_VIEW')
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [pathname])

  return null

      destroyed = true
      clearInterval(tick.current)
      try { ytPlayer.current?.destroy() } catch {}
      ytPlayer.current = null
    }
  }, [src])

  // 3. Streaming HLS (.m3u8): Safari lo reproduce directo; el resto usa hls.js
  useEffect(() => {
    if (!src || src.type !== 'hls') return
    let destroyed = false
    const v = videoRef.current
    if (!v) return
    if (v.canPlayType('application/vnd.apple.mpegurl')) {
      v.src = src.url
    } else {
      loadHls().then((Hls) => {
        if (destroyed || !Hls?.isSupported?.()) {
          if (!destroyed) setError('Tu navegador no puede reproducir este tipo de video.')
          return
        }
        const h = new Hls()
        hlsRef.current = h
        h.on(Hls.Events.ERROR, (_: any, d: any) => {
          if (d?.fatal) setError('No se pudo cargar el video (el servidor del video no permite reproducirlo aquí o el enlace venció).')
        })
        h.loadSource(src.url)
        h.attachMedia(v)
      }).catch(() => setError('No se pudo cargar el reproductor de video.'))
    }
    return () => {
      destroyed = true
      try { hlsRef.current?.destroy() } catch {}
      hlsRef.current = null
    }
  }, [src])

  // Marca de agua: cambia de lugar
  useEffect(() => {
    const t = setInterval(() => setWm({ top: 8 + Math.random() * 72, left: 4 + Math.random() * 58 }), 7000)
    return () => clearInterval(t)
  }, [])

  const pauseNow = useCallback(() => {
    const s = srcRef.current
    if (s?.type === 'youtube') ytPlayer.current?.pauseVideo?.()
    else videoRef.current?.pause()
  }, [])

  // Pausar al cambiar de pestaña
  useEffect(() => {
    const onHide = () => { if (document.hidden) pauseNow() }
    document.addEventListener('visibilitychange', onHide)
    return () => document.removeEventListener('visibilitychange', onHide)
  }, [pauseNow])

  useEffect(() => {
    const onFs = () => setFull(!!document.fullscreenElement)
    document.addEventListener('fullscreenchange', onFs)
    return () => document.removeEventListener('fullscreenchange', onFs)
  }, [])

  // ---- Seguridad: señales sospechosas -> el servidor decide si bloquea ----
  const reportSignal = useCallback((signal: string, strong: boolean) => {
    if (!srcRef.current || lockedRef.current) return
    if (strong) pauseNow()
    const now = Date.now()
    if (now - (lastSig.current[signal] || 0) < 3000) return
    lastSig.current[signal] = now
    fetch('/api/student/security', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ signal, lessonId }),
    })
      .then((r) => r.json())
      .then((d) => {
        if (d?.locked) {
          pauseNow()
          if (document.fullscreenElement) document.exitFullscreen?.().catch(() => {})
          setSrc(null)
          setPlaying(false)
          setLocked({ reason: d.reason || '' })
        }
      })
      .catch(() => {})
  }, [lessonId, pauseNow])

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      const k = e.key
      const code = e.code
      // Captura de pantalla: Impr Pant, Win+Shift+S, Cmd+Shift+3/4/5
      if (k === 'PrintScreen' || code === 'PrintScreen' ||
          (e.metaKey && e.shiftKey && (code === 'KeyS' || code === 'Digit3' || code === 'Digit4' || code === 'Digit5'))) {
        try { navigator.clipboard.writeText('') } catch {}
        reportSignal('screenshot_key', true)
        return
      }
      // Imprimir
      if ((e.ctrlKey || e.metaKey) && code === 'KeyP') { reportSignal('print', true); return }
      // Herramientas de desarrollador / ver código fuente
      if (k === 'F12' ||
          ((e.ctrlKey || e.metaKey) && e.shiftKey && (code === 'KeyI' || code === 'KeyJ' || code === 'KeyC')) ||
          (e.metaKey && e.altKey && code === 'KeyI') ||
          ((e.ctrlKey || e.metaKey) && code === 'KeyU')) {
        reportSignal('devtools_key', false)
      }
    }
    const onPrint = () => reportSignal('print', true)
    window.addEventListener('keydown', onKey)
    window.addEventListener('keyup', onKey)
    window.addEventListener('beforeprint', onPrint)
    return () => {
      window.removeEventListener('keydown', onKey)
      window.removeEventListener('keyup', onKey)
      window.removeEventListener('beforeprint', onPrint)
    }
  }, [reportSignal])

  // Mientras está bloqueado: preguntar cada 15 s si un administrador ya lo habilitó
  useEffect(() => {
    if (!locked) return
    const t = setInterval(async () => {
      try {
        const r = await fetch('/api/student/security', { cache: 'no-store' })
        const d = await r.json()
        if (r.ok && d.locked === false) { setLocked(null); setReloadKey((k) => k + 1) }
      } catch {}
    }, 15000)
    return () => clearInterval(t)
  }, [locked])

  const play = useCallback(() => {
    if (src?.type === 'youtube') ytPlayer.current?.playVideo?.()
    else videoRef.current?.play()
  }, [src])

  function pause() { pauseNow() }

  const toggle = () => (playing ? pause() : play())

  const seek = (t: number) => {
    setCur(t)
    if (src?.type === 'youtube') ytPlayer.current?.seekTo?.(t, true)
    else if (videoRef.current) videoRef.current.currentTime = t
  }

  const toggleMute = () => {
    const next = !muted
    setMuted(next)
    if (src?.type === 'youtube') next ? ytPlayer.current?.mute?.() : ytPlayer.current?.unMute?.()
    else if (videoRef.current) videoRef.current.muted = next
  }

  const cycleRate = () => {
    const next = RATES[(RATES.indexOf(rate) + 1) % RATES.length]
    setRate(next)
    if (src?.type === 'youtube') ytPlayer.current?.setPlaybackRate?.(next)
    else if (videoRef.current) videoRef.current.playbackRate = next
  }

  const toggleFull = () => {
    if (document.fullscreenElement) document.exitFullscreen()
    else boxRef.current?.requestFullscreen?.()
  }

  const watermark = src?.watermark || ''

  return (
    <div
      ref={boxRef}
      onContextMenu={(e) => e.preventDefault()}
      className="relative aspect-video w-full select-none overflow-hidden rounded-2xl bg-black"
    >
      {locked ? (
        <div className="flex h-full flex-col items-center justify-center gap-3 bg-gradient-to-br from-slate-900 to-slate-800 p-6 text-center text-white">
          <span className="flex h-16 w-16 items-center justify-center rounded-full bg-rose-500/20 text-rose-400">
            <ShieldAlert size={34} />
          </span>
          <p className="text-lg font-black">Video detenido por seguridad</p>
          <p className="max-w-md text-sm text-white/80">
            Detectamos actividad no permitida mientras veías el contenido. Tu acceso a los videos quedó suspendido hasta que el equipo realice una revisión manual.
          </p>
          {locked.reason && <p className="max-w-md rounded-lg bg-white/10 px-3 py-1.5 text-[11px] text-white/70">Motivo: {locked.reason}</p>}
          <p className="text-[11px] text-white/50">Si crees que fue un error, escríbenos por soporte. Esta pantalla se actualizará sola cuando te habiliten.</p>
        </div>
      ) : error ? (
        <div className="flex h-full flex-col items-center justify-center gap-2 p-6 text-center text-white">
          <AlertCircle size={30} className="text-rose-400" />
          <p className="text-sm font-bold">{error}</p>
        </div>
      ) : !src ? (
        <div className="flex h-full items-center justify-center text-white/70"><Loader2 className="animate-spin" /></div>
      ) : src.type === 'embed' ? (
        <>
          <iframe
            src={src.src}
            title={title}
            className="h-full w-full"
            allow="autoplay; fullscreen; picture-in-picture; encrypted-media"
            allowFullScreen
            referrerPolicy="strict-origin-when-cross-origin"
          />
          <div
            className="pointer-events-none absolute z-30 whitespace-nowrap text-[11px] font-bold text-white/30 transition-all duration-[2000ms]"
            style={{ top: `${wm.top}%`, left: `${wm.left}%`, textShadow: '0 0 4px rgba(0,0,0,.6)' }}
          >
            {watermark}
          </div>
        </>
      ) : (
        <>
          {src.type === 'youtube' ? (
            <div ref={ytHost} className="absolute inset-0 [&_iframe]:h-full [&_iframe]:w-full" />
          ) : (
            <video
              ref={videoRef}
              src={src.type === 'file' ? src.url : undefined}
              className="absolute inset-0 h-full w-full"
              controlsList="nodownload noplaybackrate noremoteplayback"
              disablePictureInPicture
              disableRemotePlayback
              playsInline
              preload="metadata"
              onLoadedMetadata={(e) => { setDur(e.currentTarget.duration); setReady(true) }}
              onCanPlay={() => setReady(true)}
              onTimeUpdate={(e) => setCur(e.currentTarget.currentTime)}
              onPlay={() => { setPlaying(true); setStarted(true) }}
              onPause={() => setPlaying(false)}
              onError={() => { if (src.type === 'file') setError('No se pudo cargar el video. Es posible que el enlace haya vencido: recarga la página.') }}
            />
          )}

          {/* Capa que bloquea cualquier clic sobre el reproductor original */}
          <div className="absolute inset-0 z-10" onClick={toggle} onDoubleClick={toggleFull} />

          {/* Portada: tapa el título y las sugerencias de YouTube mientras no se reproduce */}
          {!playing && (
            <div className="absolute inset-0 z-20 flex cursor-pointer flex-col items-center justify-center bg-gradient-to-br from-slate-900 to-slate-700 text-white" onClick={toggle}>
              {!ready ? (
                <Loader2 className="animate-spin" />
              ) : (
                <>
                  <span className="flex h-20 w-20 items-center justify-center rounded-full bg-sky-500 shadow-xl shadow-sky-500/40 transition-transform hover:scale-110">
                    <Play size={34} fill="white" className="ml-1" />
                  </span>
                  <p className="mt-4 max-w-md px-6 text-center text-sm font-bold">{started ? 'En pausa' : title}</p>
                </>
              )}
            </div>
          )}

          {/* Marca de agua con el correo del alumno */}
          <div
            className="pointer-events-none absolute z-30 whitespace-nowrap text-[11px] font-bold text-white/30 transition-all duration-[2000ms]"
            style={{ top: `${wm.top}%`, left: `${wm.left}%`, textShadow: '0 0 4px rgba(0,0,0,.6)' }}
          >
            {watermark}
          </div>

          {/* Controles propios */}
          <div className="absolute inset-x-0 bottom-0 z-40 bg-gradient-to-t from-black/85 to-transparent px-3 pb-2 pt-8" onClick={(e) => e.stopPropagation()}>
            <input
              type="range" min={0} max={dur || 0} step={0.5} value={Math.min(cur, dur || 0)}
              onChange={(e) => seek(Number(e.target.value))}
              aria-label="Progreso"
              className="h-1 w-full cursor-pointer accent-sky-400"
            />
            <div className="mt-1.5 flex items-center gap-3 text-white">
              <button type="button" onClick={toggle} aria-label={playing ? 'Pausar' : 'Reproducir'} className="cursor-pointer">
                {playing ? <Pause size={20} fill="white" /> : <Play size={20} fill="white" />}
              </button>
              <span className="text-[11px] font-bold tabular-nums">{fmt(cur)} / {fmt(dur)}</span>
              <div className="ml-auto flex items-center gap-3">
                <button type="button" onClick={cycleRate} className="rounded-md bg-white/15 px-2 py-0.5 text-[11px] font-black cursor-pointer">{rate}x</button>
                <button type="button" onClick={toggleMute} aria-label="Sonido" className="cursor-pointer">
                  {muted ? <VolumeX size={19} /> : <Volume2 size={19} />}
                </button>
                <button type="button" onClick={toggleFull} aria-label="Pantalla completa" className="cursor-pointer">
                  {full ? <Minimize size={19} /> : <Maximize size={19} />}
                </button>
              </div>
            </div>
          </div>
        </>
      )}
    </div>
  )
}

import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import bcrypt from 'bcryptjs'

export const dynamic = 'force-dynamic'

// GET - Fetch current user profile
export async function GET() {
  try {
    const session = await getServerSession(authOptions)
    if (!session || !session.user) {
      return NextResponse.json({ error: 'No autenticado' }, { status: 401 })
    }

    const userId = (session.user as any).id
    if (!userId) {
      return NextResponse.json({ error: 'ID de usuario no encontrado' }, { status: 400 })
    }

    const user = await prisma.user.findUnique({
      where: { id: userId },
      select: {
        id: true,
        name: true,
        email: true,
        phone: true,
        avatarUrl: true,
        role: true,
        status: true,
        createdAt: true,
        registeredIp: true,
        _count: {
          select: {
            progress: true,
            activityLogs: true,
          },
        },
      },
    })

    if (!user) {
      return NextResponse.json({ error: 'Usuario no encontrado' }, { status: 404 })
    }

    return NextResponse.json(user)
  } catch (error) {
    console.error('Profile fetch error:', error)
    return NextResponse.json({ error: 'Error al obtener perfil' }, { status: 500 })
  }
}

// PATCH - Update user profile (name, phone, avatarUrl, password)
export async function PATCH(req: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    if (!session || !session.user) {
      return NextResponse.json({ error: 'No autenticado' }, { status: 401 })
    }

    const userId = (session.user as any).id
    if (!userId) {
      return NextResponse.json({ error: 'ID de usuario no encontrado' }, { status: 400 })
    }

    const body = await req.json()
    const { name, phone, avatarUrl, currentPassword, newPassword } = body

    const updateData: any = {}

    if (name?.trim()) updateData.name = name.trim()
    if (phone !== undefined) updateData.phone = phone || null
    if (avatarUrl !== undefined) updateData.avatarUrl = avatarUrl || null

    // Password change flow
    if (newPassword) {
      if (!currentPassword) {
        return NextResponse.json(
          { error: 'Debes proporcionar tu contraseña actual para cambiarla.' },
          { status: 400 }
        )
      }

      if (newPassword.length < 6) {
        return NextResponse.json(
          { error: 'La nueva contraseña debe tener al menos 6 caracteres.' },
          { status: 400 }
        )
      }

      // Verify current password
      const user = await prisma.user.findUnique({ where: { id: userId }, select: { password: true } })
      if (!user) return NextResponse.json({ error: 'Usuario no encontrado' }, { status: 404 })

      const isValid = await bcrypt.compare(currentPassword, user.password)
      if (!isValid) {
        return NextResponse.json({ error: 'La contraseña actual no es correcta.' }, { status: 400 })
      }

      updateData.password = await bcrypt.hash(newPassword, 12)
    }

    if (Object.keys(updateData).length === 0) {
      return NextResponse.json({ error: 'No se proporcionaron datos para actualizar.' }, { status: 400 })
    }

    const updatedUser = await prisma.user.update({
      where: { id: userId },
      data: updateData,
      select: {
        id: true,
        name: true,
        email: true,
        phone: true,
        avatarUrl: true,
        role: true,
        status: true,
      },
    })

    return NextResponse.json({ success: true, user: updatedUser })
  } catch (error) {
    console.error('Profile update error:', error)
    return NextResponse.json({ error: 'Error al actualizar perfil' }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export const dynamic = 'force-dynamic'

// POST - Upsert lesson progress
export async function POST(req: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    if (!session || !session.user) {
      return NextResponse.json({ error: 'No autenticado' }, { status: 401 })
    }

    const userId = (session.user as any).id
    if (!userId) {
      return NextResponse.json({ error: 'ID de usuario no encontrado' }, { status: 400 })
    }

    const { lessonId, completed, watchedSecs } = await req.json()
    if (!lessonId) {
      return NextResponse.json({ error: 'lessonId requerido' }, { status: 400 })
    }

    const progress = await prisma.lessonProgress.upsert({
      where: {
        userId_lessonId: { userId, lessonId },
      },
      update: {
        completed: completed ?? false,
        watchedSecs: watchedSecs ?? 0,
      },
      create: {
        userId,
        lessonId,
        completed: completed ?? false,
        watchedSecs: watchedSecs ?? 0,
      },
    })

    return NextResponse.json({ success: true, progress })
  } catch (error) {
    console.error('Progress update error:', error)
    return NextResponse.json({ error: 'Error al actualizar progreso' }, { status: 500 })
  }
}

// GET - Get all progress for current user
export async function GET() {
  try {
    const session = await getServerSession(authOptions)
    if (!session || !session.user) {
      return NextResponse.json({ error: 'No autenticado' }, { status: 401 })
    }

    const userId = (session.user as any).id
    const progress = await prisma.lessonProgress.findMany({
      where: { userId },
      include: {
        lesson: {
          select: { id: true, title: true, moduleId: true },
        },
      },
    })

    return NextResponse.json(progress)
  } catch (error) {
    return NextResponse.json({ error: 'Error al obtener progreso' }, { status: 500 })
  }
}
